   288→function getMonday(d) {
   289→  d = new Date(d);
   290→  const day = d.getDay();
   291→  const diff = d.getDate() - day + (day === 0 ? -6 : 1);
   292→  d.setDate(diff);
   293→  d.setHours(0,0,0,0);
   294→  return d;
   295→}

   358→// GENERAZIONE TURNAZIONE
   359→// ========================
   360→function generaAssegnazioni() {
   361→  const assegnazioni = {};
   362→  const start = new Date(stato.inizioRotazione);
   363→  const gruppiIds = ['A','B','C']; // Gruppi che ruotano
   364→  const turniIds = stato.turni.filter(t => t.id !== 'fisso').map(t => t.id); // Turni senza fisso
   365→  
   366→  // Calcola quante settimane dall'inizio
   367→  const today = new Date();
   368→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 3, 0); // 3 mesi avanti
   369→  
   370→  let currentMonday = new Date(start);
   371→  let weekIndex = 0;
   372→  
   373→  while (currentMonday <= endMonth) {
   374→    // Rotazione: ogni settimana i gruppi cambiano turno
   375→    const rotIndex = weekIndex % 5; // cicla ogni 5 settimane sui 5 turni
   376→    
   377→    // Assegnazione per ogni giorno della settimana
   378→    for (let g = 0; g < 7; g++) {
   379→      const day = new Date(currentMonday);
   380→      day.setDate(day.getDate() + g);
   381→      const dk = dateKey(day);
   382→      const dayOfWeek = day.getDay(); // 0=dom, 1=lun...
   383→      
   384→      // Giorno lavorativo?
   385→      const isLavorativo = dayOfWeek >= 1 && dayOfWeek <= stato.giorniSettimana;
   386→      
   387→      if (!isLavorativo) {
   388→        assegnazioni[dk] = {A: 'riposo', B: 'riposo', C: 'riposo', J: 'riposo'};
   389→        continue;
   390→      }
   391→      
   392→      assegnazioni[dk] = {};
   393→      
   394→      // Pattern rotazione
   395→      if (stato.patternRotazione === 'A-B-C') {
   396→        // Rotazione semplice: A→B→C ogni settimana
   397→        const shift = weekIndex % 3;
   398→        const turniOrdinati = [...turniIds];
   399→        
   400→        // Ogni gruppo ha il suo turno della settimana
   401→        gruppiIds.forEach((gid, i) => {
   402→          const turnoIdx = (i + shift) % turniOrdinati.length;
   403→          assegnazioni[dk][gid] = turniOrdinati[turnoIdx];
   404→        });
   405→      } else {
   406→        // Pattern 1,2 / 2,1
   407→        const shift = weekIndex % 2 === 0 ? 0 : 1;
   408→        gruppiIds.forEach((gid, i) => {
   409→          const turnoIdx = (i + shift) % turniIds.length;
   410→          assegnazioni[dk][gid] = turniIds[turnoIdx];
   411→        });
   412→      }
   413→      
   414→      // Jolly: fisso o sostituzione
   415→      assegnazioni[dk]['J'] = 'fisso';
   416→    }
   417→    
   418→    weekIndex++;
   419→    currentMonday.setDate(currentMonday.getDate() + 7);
   420→  }
   421→  
   422→  stato.assegnazioni = assegnazioni;
   423→  salvaStato();
   424→}

   459→// CALENDARIO
   460→function renderCalendario() {
   461→  const year = meseCorrente.getFullYear();
   462→  const month = meseCorrente.getMonth();
   463→  
   464→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   465→  
   466→  const firstDay = new Date(year, month, 1);
   467→  const lastDay = new Date(year, month + 1, 0);
   468→  
   469→  // Giorno della settimana del primo (0=dom, converted to mon=0)
   470→  let startDow = firstDay.getDay() - 1;
   471→  if (startDow < 0) startDow = 6;
   472→  
   473→  const today = new Date();
   474→  
   475→  let html = '<table class="cal-table"><thead><tr>';
   476→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   477→  html += '</tr></thead><tbody>';
   478→  
   479→  let dayNum = 1;
   480→  let started = false;
   481→  
   482→  for (let row = 0; row < 6; row++) {
   483→    if (dayNum > lastDay.getDate()) break;
   484→    html += '<tr>';
   485→    for (let col = 0; col < 7; col++) {
   486→      if (!started && col < startDow) {
   487→        html += '<td></td>';
   488→      } else if (dayNum > lastDay.getDate()) {
   489→        html += '<td></td>';
   490→      } else {
   491→        started = true;
   492→        const d = new Date(year, month, dayNum);
   493→        const dk = dateKey(d);
   494→        const isToday = isSameDay(d, today);
   495→        const isHoliday = d.getDay() === 0;
   496→        const assegn = stato.assegnazioni[dk] || {};
   497→        
   498→        let cls = '';
   499→        if (isToday) cls += ' today';
   500→        if (isHoliday) cls += ' holiday';
   501→        
   502→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   503→        html += `<div class="day-num">${dayNum}</div>`;
   504→        
   505→        // Mostra turni assegnati
   506→        ['A','B','C'].forEach(gid => {
   507→          const turnoId = assegn[gid];
   508→          if (turnoId && turnoId !== 'riposo') {
   509→            const turno = stato.turni.find(t => t.id === turnoId);
   510→            if (turno) {
   511→              const gruppo = stato.gruppi.find(g => g.id === gid);
   512→              html += `<span class="day-shift" style="background:${turno.color};color:${turno.textColor}">${gid}:${turno.label}</span>`;
   513→            }
   514→          }
   515→        });
   516→        if (assegn.J === 'fisso') {
   517→          html += `<span class="day-shift turno-fisso">J:Fisso</span>`;
   518→        }
   519→        
   520→        html += '</td>';
   521→        dayNum++;
   522→      }
   523→    }
   524→    html += '</tr>';
   525→  }
   526→  html += '</tbody></table>';
   527→  
   528→  document.getElementById('calendarioGrid').innerHTML = html;
   529→}

   543→  const d = new Date(dk + 'T00:00:00');
   544→  const dayName = GIORNI_FULL[d.getDay() === 0 ? 6 : d.getDay() - 1];
   545→  document.getElementById('modalGiornoTitolo').textContent = `${dayName} ${d.getDate()} ${MESI_IT[d.getMonth()]}`;

   547→  const assegn = stato.assegnazioni[dk] || {};
   548→  let html = '';
   549→  
   550→  stato.gruppi.forEach(gruppo => {
   551→    const turnoId = assegn[gruppo.id];
   552→    const turno = turnoId ? stato.turni.find(t => t.id === turnoId) : null;
   553→    const isRiposo = turnoId === 'riposo';

   555→    html += `<div style="margin-bottom:12px">`;

   608→    html += `</div>`;

   610→    stato.gruppi.forEach(gruppo => {
   611→      const turnoId = assegn[gruppo.id];
   612→      const turno = turnoId ? stato.turni.find(t => t.id === turnoId) : null;
   613→      const isRiposo = turnoId === 'riposo';

   615→      let label = '—';

   625→      html += `<div class="turno-row" style="background:${bgColor}">`;

   630→    });

   643→// ORE
   644→function renderOre() {
   645→  const year = meseCorrente.getFullYear();
   646→  const month = meseCorrente.getMonth();
   647→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   648→  
   649→  const firstDay = new Date(year, month, 1);
   650→  const lastDay = new Date(year, month + 1, 0);
   651→  
   652→  let html = '';
   653→  
   654→  stato.gruppi.forEach(gruppo => {
   655→    gruppo.persone.forEach(persona => {
   656→      let oreLavorate = 0;
   657→      
   658→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   659→        const dk = dateKey(d);
   660→        const assegn = stato.assegnazioni[dk] || {};
   661→        const turnoId = assegn[gruppo.id];
   662→        
   663→        if (turnoId && turnoId !== 'riposo') {
   664→          oreLavorate += stato.oreTurno;
   665→        }
   666→      }
   667→      
   668→      const percentuale = gruppo.id === 'J' ? -1 : Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
   669→      const barClass = gruppo.id === 'J' ? 'ok' : (percentuale >= 100 ? 'ok' : percentuale >= 80 ? 'warn' : 'danger');
   670→      const barWidth = gruppo.id === 'J' ? 100 : percentuale;
   671→      
   672→      html += `<div class="ore-row">`;
   673→      html += `<span class="ore-nome" style="color:${gruppo.colore}">${persona}</span>`;
   674→      html += `<div style="flex:1">`;
   675→      if (gruppo.id === 'J') {
   676→        html += `<div style="font-size:12px;color:var(--text-light)">Sostituzioni — nessun monte ore fisso</div>`;
   677→      } else {
   678→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${barWidth}%"></div></div>`;
   679→        html += `<div style="display:flex;justify-content:space-between;font-size:11px;margin-top:2px">`;
   680→        html += `<span class="ore-target">Target: ${stato.oreTarget}h</span>`;
   681→        html += `<span class="ore-value" style="color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate}h (${percentuale}%)</span>`;
   682→        html += `</div>`;
   683→      }
   684→      html += `</div></div>`;
   685→    });
   686→  });
   687→  
   688→  document.getElementById('oreList').innerHTML = html;
   689→}

   724→function exportExcel() {
   725→  const year = meseCorrente.getFullYear();
   726→  const month = meseCorrente.getMonth();
   727→  const firstDay = new Date(year, month, 1);
   728→  const lastDay = new Date(year, month + 1, 0);
   729→  
   730→  let csv = 'Data,Giorno,';
   731→  stato.gruppi.forEach(g => {
   732→    g.persone.forEach(p => {
   733→      csv += p + ' (Turno),' + p + ' (Ore),';
   734→    });
   735→  });
   736→  csv += '\n';
   737→  
   738→  let totali = {};
   739→  stato.gruppi.forEach(g => g.persone.forEach(p => totali[p] = 0));
   740→  
   741→  for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   742→    const dk = dateKey(d);
   743→    const dayName = GIORNI_FULL[d.getDay() === 0 ? 6 : d.getDay() - 1];
   744→    const assegn = stato.assegnazioni[dk] || {};
   745→    
   746→    csv += `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()},${dayName},`;
   747→    
   748→    stato.gruppi.forEach(gruppo => {
   749→      const turnoId = assegn[gruppo.id];
   750→      const turno = turnoId ? stato.turni.find(t => t.id === turnoId) : null;
   751→      const isRiposo = turnoId === 'riposo';
   752→      
   753→      gruppo.persone.forEach(p => {
   754→        if (turno && !isRiposo) {
   755→          csv += turno.label + ',' + stato.oreTurno + ',';
   756→          if (gruppo.id !== 'J') totali[p] += stato.oreTurno;
   757→        } else if (isRiposo) {
   758→          csv += 'Riposo,0,';
   759→        } else {
   760→          csv += '—,0,';
   761→        }
   762→      });
   763→    });
   764→    csv += '\n';
   765→  }
   766→  
   767→  // Totali
   768→  csv += '\nTOTALI ORE,,';
   769→  stato.gruppi.forEach(g => {
   770→    g.persone.forEach(p => {
   771→      if (g.id === 'J') {
   772→        csv += ',Sostituzioni,';
   773→      } else {
   774→        csv += ',' + totali[p] + ',';
   775→      }
   776→    });
   777→  });
   778→  csv += '\n';
   779→  
   780→  // Target
   781→  csv += 'TARGET,,';
   782→  stato.gruppi.forEach(g => {
   783→    g.persone.forEach(p => {
   784→      csv += ',' + (g.id === 'J' ? '—' : stato.oreTarget) + ',';
   785→    });
   786→  });
   787→  
   788→  // Download
   789→  const blob = new Blob(['\ufeff' + csv], {type: 'text/csv;charset=utf-8;'});
   790→  const url = URL.createObjectURL(blob);
   791→  const a = document.createElement('a');
   792→  a.href = url;
   793→  a.download = `Turnazioni_${MESI_IT[month]}_${year}.csv`;
   794→  a.click();
   795→  URL.revokeObjectURL(url);
   796→}

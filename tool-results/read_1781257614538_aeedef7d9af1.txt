   776→                if (i !== riposoIdxGruppo) turnoIdx++;
   777→              }
   778→              if (turnoIdx < turniFascia.length) {
   779→                sostituzioni.push({
   780→                  persona: p,
   781→                  entrata: turniFascia[turnoIdx],
   782→                  uscita: calcolaUscita(turniFascia[turnoIdx])
   783→                });
   784→              }
   785→            }
   786→          });
   787→          assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   788→          if (sostituzioni.length > 0) {
   789→            assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;

   791→        }
   792→      }
   793→    }
   794→    
   795→    weekIndex++;
   796→    currentMonday.setDate(currentMonday.getDate() + 7);
   797→  }
   798→  
   799→  stato.assegnazioni = assegnazioni;
   800→  salvaStato();

  1062→function salvaInlineEdit(dk, persona) {
  1063→  const eInput = document.getElementById('ie_entrata');
  1064→  const uInput = document.getElementById('ie_uscita');
  1065→  if (!eInput || !uInput) return;
  1066→  
  1067→  const entrata = eInput.value.trim();
  1068→  const uscita = uInput.value.trim();

  1071→    alert('Inserisci sia entrata che uscita.');
  1072→    return;
  1073→  }
  1074→  
  1075→  const timeRegex = /^\d{1,2}:\d{2}$/;

  1077→    alert('Formato orario non valido. Usa HH:MM (es. 8:00, 15:30).');

  1081→  if (!stato.orariOverride) stato.orariOverride = {};

  1083→  
  1084→  const baseInfo = stato.assegnazioni[dk] && stato.assegnazioni[dk][persona];

  1088→      delete stato.orariOverride[dk];

  1094→  salvaStato();
  1095→  chiudiInlineEdit();
  1096→  renderCalendario();

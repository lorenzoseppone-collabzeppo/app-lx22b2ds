   429→   429→function generaAssegnazioni() {
   430→   430→  const assegnazioni = {};
   431→   431→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   432→   432→  const today = new Date();
   433→   433→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   434→   434→  
   435→   435→  const gruppi = getGruppi();
   436→   436→  const tuttiNomi = [];
   437→   437→  Object.values(gruppi).forEach(g => g.persone.forEach(p => tuttiNomi.push(p)));
   438→   438→  
   439→   439→  let currentMonday = new Date(start);
   440→   440→  let weekIndex = 0;
   441→   441→  
   442→   442→  // Traccia riposi per equità mensile
   443→   443→  const riposiMesePersona = {}; // meseKey -> persona -> count
   444→   444→  
   445→   445→  while (currentMonday <= endMonth) {
   446→   446→    for (let g = 0; g < 5; g++) { // Lun-Ven
   447→   447→      const day = new Date(currentMonday);
   448→   448→      day.setDate(day.getDate() + g);
   449→   449→      const dk = dateKey(day);
   450→   450→      const meseKey = day.getFullYear() + '-' + String(day.getMonth()+1).padStart(2,'0');
   451→   451→      
   452→   452→      assegnazioni[dk] = {};
   453→   453→      
   454→   454→      // Determina la fascia per ogni gruppo questa settimana
   455→   455→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   456→   456→        if (gid === 'J') {
   457→   457→          // Jolly: fisso 16:30-18:00 + sostituzioni
   458→   458→          assegnazioni[dk]['Jolly'] = {
   459→   459→            turno: 'fisso', entrata: '16:30', uscita: '18:00',
   460→   460→            ore: 1.5, stato: 'lavoro', gruppo: 'J'
   461→   461→          };
   462→   462→          return;
   463→   463→        }
   464→   464→        
   465→   465→        const fascia = getFasciaForWeek(weekIndex, gid);
   466→   466→        const turniFascia = FASCE_TURNI[fascia];
   467→   467→        
   468→   468→        gruppo.persone.forEach((persona, idx) => {
   469→   469→          if (idx < turniFascia.length) {
   470→   470→            const entrata = turniFascia[idx];
   471→   471→            const uscita = calcolaUscita(entrata);
   472→   472→            assegnazioni[dk][persona] = {
   473→   473→              turno: entrata, entrata: entrata, uscita: uscita,
   474→   474→              ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   475→   475→            };
   476→   476→          } else {
   477→   477→            // Terza persona → riposo o reperibile
   478→   478→            assegnazioni[dk][persona] = {
   479→   479→              turno: 'riposo', entrata: null, uscita: null,
   480→   480→              ore: 0, stato: 'riposo', gruppo: gid
   481→   481→            };
   482→   482→          }
   483→   483→        });
   484→   484→      });
   485→   485→      
   486→   486→      // SISTEMA RIPOSI: 2 persone a riposo, da gruppi diversi
   487→   487→      // 1 delle 2 è reperibile
   488→   488→      const personeRiposo = [];
   489→   489→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   490→   490→        gruppo.persone.forEach(persona => {
   491→   491→          if (assegnazioni[dk][persona] && assegnazioni[dk][persona].stato === 'riposo') {
   492→   492→            personeRiposo.push(persona);
   493→   493→          }
   494→   494→        });
   495→   495→      });
   496→   496→      
   497→   497→      // Se ci sono già 2+ persone a riposo, ok
   498→   498→      // Se ce n'è solo 1, aggiungiamo 1 persona da un altro gruppo
   499→   499→      if (personeRiposo.length < stato.giorniRiposo) {
   500→   500→        // Trova chi può andare a riposo (da gruppi diversi da chi è già a riposo)
   501→   501→        const gruppiGiaRiposo = new Set(personeRiposo.map(p => assegnazioni[dk][p].gruppo));
   502→   502→        
   503→   503→        // Scegli chi ha riposato di meno questo mese
   504→   504→        const conteggio = {};
   505→   505→        tuttiNomi.forEach(p => conteggio[p] = 0);
   506→   506→        Object.keys(assegnazioni).forEach(rdk => {
   507→   507→          if (rdk.startsWith(meseKey) && assegnazioni[rdk]) {
   508→   508→            Object.entries(assegnazioni[rdk]).forEach(([persona, info]) => {
   509→   509→              if (info.stato === 'riposo' && conteggio[persona] !== undefined) conteggio[persona]++;
   510→   510→            });
   511→   511→          }
   512→   512→        });
   513→   513→        
   514→   514→        const candidati = tuttiNomi.filter(p => {
   515→   515→          if (personeRiposo.includes(p)) return false;
   516→   516→          if (p === 'Jolly') return false;
   517→   517→          const grp = getGruppoPersona(p);
   518→   518→          return !gruppiGiaRiposo.has(grp);
   519→   519→        }).sort((a,b) => conteggio[a] - conteggio[b]);
   520→   520→        
   521→   521→        while (personeRiposo.length < stato.giorniRiposo && candidati.length > 0) {
   522→   522→          const persona = candidati.shift();
   523→   523→          const gruppoOld = assegnazioni[dk][persona];
   524→   524→          const nuovoGruppoRiposo = getGruppoPersona(persona);
   525→   525→          
   526→   526→          // Metti a riposo
   527→   527→          assegnazioni[dk][persona] = {
   528→   528→            turno: 'riposo', entrata: null, uscita: null,
   529→   529→            ore: 0, stato: 'riposo', gruppo: nuovoGruppoRiposo
   530→   530→          };
   531→   531→          personeRiposo.push(persona);
   532→   532→          
   533→   533→          // Aggiungi il gruppo ai già in riposo
   534→   534→          gruppiGiaRiposo.add(nuovoGruppoRiposo);
   535→   535→        }
   536→   536→      }
   537→   537→      
   538→   538→      // Segna 1 riposo come reperibile
   539→   539→      if (personeRiposo.length >= 2) {
   540→   540→        // Il secondo diventa reperibile
   541→   541→        assegnazioni[dk][personeRiposo[1]] = {
   542→   542→          ...assegnazioni[dk][personeRiposo[1]],
   543→   543→          stato: 'reperibile',
   544→   544→          turno: 'reperibile'
   545→   545→        };
   546→   546→      }
   547→   547→      
   548→   548→      // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   549→   549→      if (gruppi.J && personeRiposo.length > 0) {
   550→   550→        const sostituzioni = [];
   551→   551→        personeRiposo.forEach(p => {
   552→   552→          // Trova il turno che avrebbe dovuto fare
   553→   553→          const gid = getGruppoPersona(p);
   554→   554→          if (gid && gid !== 'J') {
   555→   555→            const fascia = getFasciaForWeek(weekIndex, gid);
   556→   556→            const turniFascia = FASCE_TURNI[fascia];
   557→   557→            const idx = gruppi[gid].persone.indexOf(p);
   558→   558→            if (idx >= 0 && idx < turniFascia.length) {
   559→   559→              sostituzioni.push({
   560→   560→                persona: p,
   561→   561→                entrata: turniFascia[idx],
   562→   562→                uscita: calcolaUscita(turniFascia[idx])
   563→   563→              });
   564→   564→            }
   565→   565→          }
   566→   566→        });
   567→   567→        assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   568→   568→        if (sostituzioni.length > 0) {
   569→   569→          assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   570→   570→        }
   571→   571→      }
   572→   572→    }
   573→   573→    
   574→   574→    weekIndex++;
   575→   575→    currentMonday.setDate(currentMonday.getDate() + 7);
   576→   576→  }
   577→   577→  
   578→   578→  stato.assegnazioni = assegnazioni;
   579→   579→  salvaStato();
   580→   580→}

   617→   617→function renderCalendario() {
   618→   618→  const year = meseCorrente.getFullYear();
   619→   619→  const month = meseCorrente.getMonth();
   620→   620→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   621→   621→  
   622→   622→  const firstDay = new Date(year, month, 1);
   623→   623→  const lastDay = new Date(year, month + 1, 0);
   624→   624→  let startDow = firstDay.getDay() - 1;
   625→   625→  if (startDow < 0) startDow = 6;
   626→   626→  const today = new Date();
   627→   627→  
   628→   628→  let html = '<table class="cal-table"><thead><tr>';
   629→   629→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   630→   630→  html += '</tr></thead><tbody>';
   631→   631→  
   632→   632→  let dayNum = 1, started = false;
   633→   633→  for (let row = 0; row < 6; row++) {
   634→   634→    if (dayNum > lastDay.getDate()) break;
   635→   635→    html += '<tr>';
   636→   636→    for (let col = 0; col < 7; col++) {
   637→   637→      if (!started && col < startDow) { html += '<td></td>'; }
   638→   638→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   639→   639→      else {
   640→   640→        started = true;
   641→   641→        const d = new Date(year, month, dayNum);
   642→   642→        const dk = dateKey(d);
   643→   643→        const isToday = isSameDay(d, today);
   644→   644→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;
   645→   645→        
   646→   646→        let cls = '';
   647→   647→        if (isToday) cls += ' today';
   648→   648→        if (isWeekend) cls += ' weekend';
   649→   649→        
   650→   650→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   651→   651→        html += `<div class="day-num">${dayNum}</div>`;
   652→   652→        
   653→   653→        if (!isWeekend) {
   654→   654→          const ass = stato.assegnazioni[dk] || {};
   655→   655→          // Mostra ingressi per gruppo
   656→   656→          Object.entries(getGruppi()).forEach(([gid, gruppo]) => {

   663→   663→                else if (ass[p].stato === 'riposo') turni.push('R');
   664→   664→                else if (ass[p].stato === 'reperibile') turni.push('r');
   665→   665→              }
   666→   666→            });

   668→   668→            const colore = turni[0] && TURNI_COLORS[turni[0]] ? TURNI_COLORS[turni[0]].bg : '#f1f5f9';
   669→   669→            const coloreTx = turni[0] && TURNI_COLORS[turni[0]] ? TURNI_COLORS[turni[0]].tx : '#475569';

   879→   879→function renderOre() {
   880→   880→  const year = meseCorrente.getFullYear();
   881→   881→  const month = meseCorrente.getMonth();
   882→   882→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   883→   883→  
   884→   884→  const firstDay = new Date(year, month, 1);
   885→   885→  const lastDay = new Date(year, month + 1, 0);
   886→   886→  const gruppi = getGruppi();
   887→   887→  
   888→   888→  let html = '';
   889→   889→  
   890→   890→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   891→   891→    gruppo.persone.forEach(persona => {
   892→   892→      let oreLavorate = 0, giorniLavorati = 0, giorniRiposo = 0;
   893→   893→      
   894→   894→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   895→   895→        const dk = dateKey(d);
   896→   896→        const dow = d.getDay();
   897→   897→        if (dow === 0 || dow === 6) continue;
   898→   898→        
   899→   899→        const ass = stato.assegnazioni[dk];
   900→   900→        if (!ass || !ass[persona]) continue;
   901→   901→        
   902→   902→        const info = ass[persona];
   903→   903→        if (info.stato === 'lavoro') {
   904→   904→          oreLavorate += info.ore || stato.oreTurno;
   905→   905→          giorniLavorati++;
   906→   906→        } else if (info.stato === 'riposo' || info.stato === 'reperibile') {
   907→   907→          giorniRiposo++;
   908→   908→        }
   909→   909→      }
   910→   910→      
   911→   911→      if (gid === 'J') {
   912→   912→        html += `<div class="ore-row">`;
   913→   913→        html += `<span class="gruppo-badge gruppo-J" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   914→   914→        html += `<div style="flex:1">`;
   915→   915→        html += `<div style="font-size:12px">Fisso + sostituzioni: <strong>${oreLavorate.toFixed(1)}h</strong> (${giorniLavorati}gg)</div>`;
   916→   916→        html += `</div></div>`;
   917→   917→      } else {
   918→   918→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
   919→   919→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
   920→   920→        const diff = oreLavorate - stato.oreTarget;
   921→   921→        
   922→   922→        html += `<div class="ore-row">`;
   923→   923→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   924→   924→        html += `<div style="flex:1">`;
   925→   925→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
   926→   926→        html += `<span>${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
   927→   927→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
   928→   928→        html += `</div>`;
   929→   929→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
   930→   930→        html += `</div></div>`;
   931→   931→      }
   932→   932→    });
   933→   933→  });

   550→function generaAssegnazioni() {
   551→  const assegnazioni = {};
   552→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   553→  const today = new Date();
   554→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   555→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   556→  
   557→  const gruppi = getGruppi();
   558→  const isVar9 = stato.variante === 2;
   559→  
   560→  // Variante 2: lista piatta di tutte le 9 persone per la rotazione globale
   561→  let rotazioneGlobale = [];
   562→  if (isVar9) {
   563→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   564→      if (gid === 'J') return;
   565→      gruppo.persone.forEach(p => rotazioneGlobale.push({persona: p, gruppo: gid}));
   566→    });
   567→  }
   568→  
   569→  let currentMonday = getMonday(startMonth);
   570→  const startMonday = getMonday(start);
   571→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   572→  let weekIndex = weeksDiffFromStart;
   573→  
   574→  while (currentMonday <= endMonth) {
   575→    for (let g = 0; g < 5; g++) { // Lun-Ven
   576→      const day = new Date(currentMonday);
   577→      day.setDate(day.getDate() + g);
   578→      const dk = dateKey(day);
   579→      
   580→      assegnazioni[dk] = {};
   581→      const personeRiposoOggi = [];
   582→      
   583→      const globalDayIdx = weekIndex * 5 + g;
   584→      
   585→      if (isVar9) {
   586→        const riposoIdx = ((globalDayIdx % 9) + 9) % 9;
   587→        const chiRiposa = rotazioneGlobale[riposoIdx];
   588→        
   589→        const fasciaRiposo = getFasciaForWeek(weekIndex, chiRiposa.gruppo);
   590→        const turniFasciaRiposo = FASCE_TURNI[fasciaRiposo];
   591→        const jollyIngresso = turniFasciaRiposo[0];
   592→        
   593→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   594→          if (gid === 'J') {
   595→            assegnazioni[dk]['Jolly'] = {
   596→              turno: 'jolly', entrata: jollyIngresso, uscita: calcolaUscita(jollyIngresso),
   597→              ore: stato.oreTurno, stato: 'lavoro', gruppo: 'J',
   598→              fisso: '16:30-18:00',
   599→              copre: chiRiposa.persona,
   600→              copreGruppo: chiRiposa.gruppo,
   601→              oreTotali: stato.oreTurno + 1.5
   602→            };
   603→            return;
   604→          }
   605→          
   606→          const fascia = getFasciaForWeek(weekIndex, gid);
   607→          const turniFascia = FASCE_TURNI[fascia];
   608→          
   609→          let workingCount = 0;
   610→          gruppo.persone.forEach((persona) => {
   611→            if (persona === chiRiposa.persona && gid === chiRiposa.gruppo) {
   612→              assegnazioni[dk][persona] = {
   613→                turno: 'riposo', entrata: null, uscita: null,
   614→                ore: 0, stato: 'riposo', gruppo: gid
   615→              };
   616→              personeRiposoOggi.push(persona);
   617→            } else {
   618→              const entrataIdx = workingCount % turniFascia.length;
   619→              const entrata = turniFascia[entrataIdx];
   620→              const uscita = calcolaUscita(entrata);
   621→              assegnazioni[dk][persona] = {
   622→                turno: entrata, entrata: entrata, uscita: uscita,
   623→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   624→              };
   625→              workingCount++;
   626→            }
   627→          });
   628→        });
   629→        
   630→      } else {
   631→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   632→          if (gid === 'J') {
   633→            assegnazioni[dk]['Jolly'] = {
   634→              turno: 'fisso', entrata: '16:30', uscita: '18:00',
   635→              ore: 1.5, stato: 'lavoro', gruppo: 'J'
   636→            };
   637→            return;
   638→          }
   639→          
   640→          const fascia = getFasciaForWeek(weekIndex, gid);
   641→          const turniFascia = FASCE_TURNI[fascia];
   642→          const numPersone = gruppo.persone.length;
   643→          const numIngressi = turniFascia.length;
   644→          const haRiposo = numPersone > numIngressi;
   645→          
   646→          if (haRiposo) {
   647→            const riposoIdx = ((globalDayIdx % numPersone) + numPersone) % numPersone;
   648→            let turnoAssegnato = 0;
   649→            gruppo.persone.forEach((persona, idx) => {
   650→              if (idx === riposoIdx) {
   651→                assegnazioni[dk][persona] = {
   652→                  turno: 'riposo', entrata: null, uscita: null,
   653→                  ore: 0, stato: 'riposo', gruppo: gid
   654→                };
   655→                personeRiposoOggi.push(persona);
   656→              } else {
   657→                const entrata = turniFascia[turnoAssegnato];
   658→                const uscita = calcolaUscita(entrata);
   659→                assegnazioni[dk][persona] = {
   660→                  turno: entrata, entrata: entrata, uscita: uscita,
   661→                  ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   662→                };
   663→                turnoAssegnato++;
   664→              }
   665→            });
   666→          } else {
   667→            gruppo.persone.forEach((persona, idx) => {
   668→              const entrata = turniFascia[idx];
   669→              const uscita = calcolaUscita(entrata);
   670→              assegnazioni[dk][persona] = {
   671→                turno: entrata, entrata: entrata, uscita: uscita,
   672→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   673→              };
   674→            });
   675→          }
   676→        });
   677→        
   678→        // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   679→        if (gruppi.J && personeRiposoOggi.length > 0) {
   680→          const sostituzioni = [];
   681→          personeRiposoOggi.forEach(p => {
   682→            const gid = getGruppoPersona(p);
   683→            if (gid && gid !== 'J') {
   684→              const fascia = getFasciaForWeek(weekIndex, gid);
   685→              const turniFascia = FASCE_TURNI[fascia];
   686→              const pIdx = gruppi[gid].persone.indexOf(p);
   687→              let turnoIdx = 0;
   688→              const riposoIdxGruppo = ((globalDayIdx % gruppi[gid].persone.length) + gruppi[gid].persone.length) % gruppi[gid].persone.length;
   689→              for (let i = 0; i < pIdx; i++) {
   690→                if (i !== riposoIdxGruppo) turnoIdx++;
   691→              }
   692→              if (turnoIdx < turniFascia.length) {
   693→                sostituzioni.push({
   694→                  persona: p,
   695→                  entrata: turniFascia[turnoIdx],
   696→                  uscita: calcolaUscita(turniFascia[turnoIdx])
   697→                });
   698→              }
   699→            }
   700→          });
   701→          assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   702→          if (sostituzioni.length > 0) {
   703→            assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   704→          }
   705→        }
   706→      }
   707→    }
   708→    
   709→    weekIndex++;
   710→    currentMonday.setDate(currentMonday.getDate() + 7);
   711→  }
   712→  
   713→  stato.assegnazioni = assegnazioni;
   714→  salvaStato();
   715→}

   784→function renderCalendario() {
   785→  const year = meseCorrente.getFullYear();
   786→  const month = meseCorrente.getMonth();
   787→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   788→  
   789→  const firstDay = new Date(year, month, 1);
   790→  const lastDay = new Date(year, month + 1, 0);
   791→  let startDow = firstDay.getDay() - 1;
   792→  if (startDow < 0) startDow = 6;
   793→  const today = new Date();
   794→  const gruppi = getGruppi();
   795→  
   796→  let html = '<table class="cal-table"><thead><tr>';
   797→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   798→  html += '</tr></thead><tbody>';
   799→  
   800→  let dayNum = 1, started = false;
   801→  for (let row = 0; row < 6; row++) {
   802→    if (dayNum > lastDay.getDate()) break;
   803→    html += '<tr>';
   804→    for (let col = 0; col < 7; col++) {
   805→      if (!started && col < startDow) { html += '<td></td>'; }
   806→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   807→      else {
   808→        started = true;
   809→        const d = new Date(year, month, dayNum);
   810→        const dk = dateKey(d);
   811→        const isToday = isSameDay(d, today);
   812→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;
   813→        
   814→        let cls = '';
   815→        if (isToday) cls += ' today';
   816→        if (isWeekend) cls += ' weekend';
   817→        
   818→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   819→        html += `<div class="day-num">${dayNum}</div>`;
   820→        
   821→        if (!isWeekend) {
   822→          const ass = stato.assegnazioni[dk] || {};
   823→          const hasData = Object.keys(ass).length > 0;
   824→          
   825→          if (hasData) {
   826→            Object.entries(gruppi).forEach(([gid, gruppo], grpIdx) => {
   827→              if (grpIdx > 0) html += '<div class="cal-grp-sep"></div>';
   828→              gruppo.persone.forEach(persona => {
   829→                const info = getTurnoEffettivo(dk, persona);
   830→                if (!info) return;
   831→                
   832→                const dotColor = GRUPPO_COLORS[gid] || '#94a3b8';
   833→                const isOverridden = info.overridden;
   834→                const personClass = 'cal-person' + (editMode ? ' editing' : '') + (isOverridden ? ' cal-person-overridden' : '');
   835→                html += `<div class="${personClass}" ${editMode ? `onclick="event.stopPropagation();mostraGiorno('${dk}')"` : ''}>`;
   836→                html += `<span class="cal-person-dot" style="background:${dotColor}"></span>`;
   837→                
   838→                const shortName = getNomeBreve(persona);
   839→                html += `<span class="cal-person-name" style="color:${dotColor}" title="${getNomePersona(persona)}">${shortName}</span>`;
   840→                
   841→                if (info.stato === 'lavoro' && info.entrata) {
   842→                  if (info.turno === 'jolly') {

   992→// Save override for a specific person on a specific day
   993→function salvaOverride(dk, persona) {
   994→  const eInput = document.getElementById('edit_e_' + dk + '_' + persona);
   995→  const uInput = document.getElementById('edit_u_' + dk + '_' + persona);
   996→  if (!eInput || !uInput) return;
   997→  
   998→  const entrata = eInput.value.trim();
   999→  const uscita = uInput.value.trim();
  1000→  
  1001→  if (!entrata || !uscita) {
  1002→    alert('Inserisci sia entrata che uscita.');
  1003→    return;
  1004→  }
  1005→  
  1006→  // Validate time format
  1007→  const timeRegex = /^\d{1,2}:\d{2}$/;
  1008→  if (!timeRegex.test(entrata) || !timeRegex.test(uscita)) {
  1009→    alert('Formato orario non valido. Usa HH:MM (es. 8:00, 15:30).');
  1010→    return;
  1011→  }
  1012→  
  1013→  if (!stato.orariOverride) stato.orariOverride = {};
  1014→  if (!stato.orariOverride[dk]) stato.orariOverride[dk] = {};
  1015→  
  1016→  // Check if the override is same as original → remove it
  1017→  const baseInfo = stato.assegnazioni[dk] && stato.assegnazioni[dk][persona];
  1018→  if (baseInfo && baseInfo.entrata === entrata && baseInfo.uscita === uscita) {
  1019→    delete stato.orariOverride[dk][persona];
  1020→    if (Object.keys(stato.orariOverride[dk]).length === 0) {
  1021→      delete stato.orariOverride[dk];
  1022→    }
  1023→  } else {
  1024→    stato.orariOverride[dk][persona] = { entrata, uscita };
  1025→  }
  1026→  
  1027→  salvaStato();
  1028→  // Re-render the modal with updated data
  1029→  mostraGiorno(dk);
  1030→  renderCalendario();
  1031→}

  1231→        html += `</div></div>`;
  1232→      } else {
  1233→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
  1234→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
  1235→        const diff = oreLavorate - stato.oreTarget;
  1236→        
  1237→        html += `<div class="ore-row">`;
  1238→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${nomeMostrato.substring(0,2)}</span>`;
  1239→        html += `<div style="flex:1">`;
  1240→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
  1241→        html += `<span>${nomeMostrato} — ${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
  1242→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
  1243→        html += `</div>`;
  1244→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
  1245→        html += `</div></div>`;
  1246→      }
  1247→    });
  1248→  });
  1249→  
  1250→  document.getElementById('oreList').innerHTML = html;
  1251→}

  1267→      const nome = getNomePersona(p);

  1269→    });
  1270→  });
  1271→  csv += 'Presenti 9:30-15:30,\n';
  1272→  
  1273→  let totali = {};

  1275→  
  1276→  for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
  1277→    const dk = dateKey(d);
  1278→    const dow = d.getDay();
  1279→    if (dow === 0 || dow === 6) continue;
  1280→    
  1281→    const dayName = GIORNI_FULL[dow === 0 ? 6 : dow - 1];
  1282→    
  1283→    csv += `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()},${dayName},`;
  1284→    
  1285→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
  1286→      gruppo.persone.forEach(persona => {
  1287→        const info = getTurnoEffettivo(dk, persona);

  1289→        
  1290→        csv += `${info.entrata},${info.entrata}-${info.uscita},${info.ore.toFixed(1)},`;
  1291→        totali[persona] += info.ore || stato.oreTurno;

  1304→      }

  1307→  }

  1309→  // Totali
  1310→  csv += '\nTOTALI ORE,,';

  1313→      csv += ',,' + totali[p].toFixed(1) + ',';
  1314→    });
  1315→  });

  1317→

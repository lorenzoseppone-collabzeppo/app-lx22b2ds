   430→   430→function generaAssegnazioni() {
   431→   431→  const assegnazioni = {};
   432→   432→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   433→   433→  const today = new Date();
   434→   434→  // Genera dal mese precedente fino a 6 mesi avanti per coprire giorni prima dell'inizio rotazione
   435→   435→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   436→   436→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   437→   437→  
   438→   438→  const gruppi = getGruppi();
   439→   439→  const tuttiNomi = [];
   440→   440→  Object.values(gruppi).forEach(g => g.persone.forEach(p => tuttiNomi.push(p)));
   441→   441→  
   442→   442→  // Calcola il weekIndex basato sulla data di inizio rotazione
   443→   443→  // Partiamo da prima del mese corrente per generare anche i giorni precedenti
   444→   444→  let currentMonday = getMonday(startMonth);
   445→   445→  // Il weekIndex relativo alla data di inizio rotazione
   446→   446→  const startMonday = getMonday(start);
   447→   447→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   448→   448→  let weekIndex = weeksDiffFromStart;
   449→   449→  
   450→   450→  // Conta i giorni lavorativi passati per ruotare i riposi
   451→   451→  let giornoLavorativoIdx = 0;
   452→   452→  
   453→   453→  while (currentMonday <= endMonth) {
   454→   454→    for (let g = 0; g < 5; g++) { // Lun-Ven
   455→   455→      const day = new Date(currentMonday);
   456→   456→      day.setDate(day.getDate() + g);
   457→   457→      const dk = dateKey(day);
   458→   458→      const meseKey = day.getFullYear() + '-' + String(day.getMonth()+1).padStart(2,'0');
   459→   459→      
   460→   460→      assegnazioni[dk] = {};
   461→   461→      const personeRiposoOggi = [];
   462→   462→      
   463→   463→      // Determina la fascia per ogni gruppo questa settimana
   464→   464→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   465→   465→        if (gid === 'J') {
   466→   466→          // Jolly: fisso 16:30-18:00 + sostituzioni
   467→   467→          assegnazioni[dk]['Jolly'] = {
   468→   468→            turno: 'fisso', entrata: '16:30', uscita: '18:00',
   469→   469→            ore: 1.5, stato: 'lavoro', gruppo: 'J'
   470→   470→          };
   471→   471→          return;
   472→   472→        }
   473→   473→        
   474→   474→        const fascia = getFasciaForWeek(weekIndex, gid);
   475→   475→        const turniFascia = FASCE_TURNI[fascia]; // 2 ingressi
   476→   476→        const numPersone = gruppo.persone.length;
   477→   477→        const numIngressi = turniFascia.length; // sempre 2
   478→   478→        
   479→   479→        // Il riposo ruota SOLO nei gruppi che hanno più persone che ingressi
   480→   480→        // Es: gruppo B (3 persone, 2 ingressi) → 1 persona a riposo che ruota
   481→   481→        // Es: gruppo A (2 persone, 2 ingressi) → nessun riposo, tutti lavorano
   482→   482→        const haRiposo = numPersone > numIngressi;
   483→   483→        
   484→   484→        if (haRiposo) {
   485→   485→          // IL RIPOSO RUOTA: ogni giorno tocca a una persona diversa
   486→   486→          const riposoIdx = giornoLavorativoIdx % numPersone;
   487→   487→          
   488→   488→          let turnoAssegnato = 0;
   489→   489→          gruppo.persone.forEach((persona, idx) => {
   490→   490→            if (idx === riposoIdx) {
   491→   491→              assegnazioni[dk][persona] = {
   492→   492→                turno: 'riposo', entrata: null, uscita: null,
   493→   493→                ore: 0, stato: 'riposo', gruppo: gid
   494→   494→              };
   495→   495→              personeRiposoOggi.push(persona);
   496→   496→            } else {
   497→   497→              const entrata = turniFascia[turnoAssegnato];
   498→   498→              const uscita = calcolaUscita(entrata);
   499→   499→              assegnazioni[dk][persona] = {
   500→   500→                turno: entrata, entrata: entrata, uscita: uscita,
   501→   501→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   502→   502→              };
   503→   503→              turnoAssegnato++;
   504→   504→            }
   505→   505→          });
   506→   506→        } else {
   507→   507→          // Gruppo con stesso numero di persone e ingressi → tutti lavorano
   508→   508→          gruppo.persone.forEach((persona, idx) => {
   509→   509→            const entrata = turniFascia[idx];
   510→   510→            const uscita = calcolaUscita(entrata);
   511→   511→            assegnazioni[dk][persona] = {
   512→   512→              turno: entrata, entrata: entrata, uscita: uscita,
   513→   513→              ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   514→   514→            };
   515→   515→          });
   516→   516→        }
   517→   517→      });
   518→   518→      
   519→   519→      // 1 delle persone a riposo diventa reperibile
   520→   520→      if (personeRiposoOggi.length >= 2) {
   521→   521→        // Il secondo riposo diventa reperibile
   522→   522→        assegnazioni[dk][personeRiposoOggi[1]] = {
   523→   523→          ...assegnazioni[dk][personeRiposoOggi[1]],
   524→   524→          stato: 'reperibile',
   525→   525→          turno: 'reperibile'
   526→   526→        };
   527→   527→      }
   528→   528→      
   529→   529→      // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   530→   530→      if (gruppi.J && personeRiposoOggi.length > 0) {
   531→   531→        const sostituzioni = [];
   532→   532→        personeRiposoOggi.forEach(p => {
   533→   533→          const gid = getGruppoPersona(p);
   534→   534→          if (gid && gid !== 'J') {
   535→   535→            const fascia = getFasciaForWeek(weekIndex, gid);
   536→   536→            const turniFascia = FASCE_TURNI[fascia];
   537→   537→            // Trova quale turno avrebbe fatto (in base alla posizione nel gruppo senza il riposo)
   538→   538→            const pIdx = gruppi[gid].persone.indexOf(p);
   539→   539→            let turnoIdx = 0;
   540→   540→            for (let i = 0; i < pIdx; i++) {
   541→   541→              if (i !== (giornoLavorativoIdx % gruppi[gid].persone.length)) turnoIdx++;
   542→   542→            }
   543→   543→            if (turnoIdx < turniFascia.length) {
   544→   544→              sostituzioni.push({
   545→   545→                persona: p,
   546→   546→                entrata: turniFascia[turnoIdx],
   547→   547→                uscita: calcolaUscita(turniFascia[turnoIdx])
   548→   548→              });
   549→   549→            }
   550→   550→          }
   551→   551→        });
   552→   552→        assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   553→   553→        if (sostituzioni.length > 0) {
   554→   554→          assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   555→   555→        }
   556→   556→      }
   557→   557→      
   558→   558→      giornoLavorativoIdx++;
   559→   559→    }
   560→   560→    
   561→   561→    weekIndex++;
   562→   562→    currentMonday.setDate(currentMonday.getDate() + 7);
   563→   563→  }
   564→   564→  
   565→   565→  stato.assegnazioni = assegnazioni;
   566→   566→  salvaStato();
   567→   567→}

   606→   606→function renderCalendario() {
   607→   607→  const year = meseCorrente.getFullYear();
   608→   608→  const month = meseCorrente.getMonth();
   609→   609→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   610→   610→  
   611→   611→  const firstDay = new Date(year, month, 1);
   612→   612→  const lastDay = new Date(year, month + 1, 0);
   613→   613→  let startDow = firstDay.getDay() - 1;
   614→   614→  if (startDow < 0) startDow = 6;
   615→   615→  const today = new Date();
   616→   616→  const gruppi = getGruppi();
   617→   617→  
   618→   618→  let html = '<table class="cal-table"><thead><tr>';
   619→   619→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   620→   620→  html += '</tr></thead><tbody>';
   621→   621→  
   622→   622→  let dayNum = 1, started = false;
   623→   623→  for (let row = 0; row < 6; row++) {
   624→   624→    if (dayNum > lastDay.getDate()) break;
   625→   625→    html += '<tr>';
   626→   626→    for (let col = 0; col < 7; col++) {
   627→   627→      if (!started && col < startDow) { html += '<td></td>'; }
   628→   628→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   629→   629→      else {
   630→   630→        started = true;
   631→   631→        const d = new Date(year, month, dayNum);
   632→   632→        const dk = dateKey(d);
   633→   633→        const isToday = isSameDay(d, today);
   634→   634→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;

   882→   882→function renderOre() {
   883→   883→  const year = meseCorrente.getFullYear();
   884→   884→  const month = meseCorrente.getMonth();
   885→   885→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   886→   886→  
   887→   887→  const firstDay = new Date(year, month, 1);
   888→   888→  const lastDay = new Date(year, month + 1, 0);
   889→   889→  const gruppi = getGruppi();
   890→   890→  
   891→   891→  let html = '';
   892→   892→  
   893→   893→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   894→   894→    gruppo.persone.forEach(persona => {
   895→   895→      let oreLavorate = 0, giorniLavorati = 0, giorniRiposo = 0;
   896→   896→      
   897→   897→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   898→   898→        const dk = dateKey(d);
   899→   899→        const dow = d.getDay();
   900→   900→        if (dow === 0 || dow === 6) continue;
   901→   901→        
   902→   902→        const ass = stato.assegnazioni[dk];
   903→   903→        if (!ass || !ass[persona]) continue;
   904→   904→        
   905→   905→        const info = ass[persona];
   906→   906→        if (info.stato === 'lavoro') {
   907→   907→          oreLavorate += info.ore || stato.oreTurno;
   908→   908→          giorniLavorati++;
   909→   909→        } else if (info.stato === 'riposo' || info.stato === 'reperibile') {
   910→   910→          giorniRiposo++;
   911→   911→        }
   912→   912→      }
   913→   913→      
   914→   914→      if (gid === 'J') {
   915→   915→        html += `<div class="ore-row">`;
   916→   916→        html += `<span class="gruppo-badge gruppo-J" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   917→   917→        html += `<div style="flex:1">`;
   918→   918→        html += `<div style="font-size:12px">Fisso + sostituzioni: <strong>${oreLavorate.toFixed(1)}h</strong> (${giorniLavorati}gg)</div>`;
   919→   919→        html += `</div></div>`;
   920→   920→      } else {
   921→   921→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
   922→   922→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
   923→   923→        const diff = oreLavorate - stato.oreTarget;
   924→   924→        
   925→   925→        html += `<div class="ore-row">`;
   926→   926→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   927→   927→        html += `<div style="flex:1">`;
   928→   928→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
   929→   929→        html += `<span>${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
   930→   930→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
   931→   931→        html += `</div>`;
   932→   932→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
   933→   933→        html += `</div></div>`;
   934→   934→      }
   935→   935→    });
   936→   936→  });

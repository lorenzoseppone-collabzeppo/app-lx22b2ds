   348→function getMonday(d) {
   349→  d = new Date(d);
   350→  const day = d.getDay();
   351→  const diff = d.getDate() - day + (day === 0 ? -6 : 1);
   352→  d.setDate(diff);
   353→  d.setHours(0,0,0,0);
   354→  return d;
   355→}

   357→function dateKey(d) {
   358→  const y = d.getFullYear();
   359→  const m = String(d.getMonth()+1).padStart(2,'0');
   360→  const dd = String(d.getDate()).padStart(2,'0');
   361→  return `${y}-${m}-${dd}`;
   362→}

   430→function generaAssegnazioni() {
   431→  const assegnazioni = {};
   432→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   433→  const today = new Date();
   434→  // Genera dal mese precedente fino a 6 mesi avanti per coprire giorni prima dell'inizio rotazione
   435→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   436→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   437→  
   438→  const gruppi = getGruppi();
   439→  const tuttiNomi = [];
   440→  Object.values(gruppi).forEach(g => g.persone.forEach(p => tuttiNomi.push(p)));
   441→  
   442→  // Calcola il weekIndex basato sulla data di inizio rotazione
   443→  // Partiamo da prima del mese corrente per generare anche i giorni precedenti
   444→  let currentMonday = getMonday(startMonth);
   445→  // Il weekIndex relativo alla data di inizio rotazione
   446→  const startMonday = getMonday(start);
   447→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   448→  let weekIndex = weeksDiffFromStart;
   449→  
   450→  // Conta i giorni lavorativi passati per ruotare i riposi
   451→  let giornoLavorativoIdx = 0;
   452→  
   453→  while (currentMonday <= endMonth) {
   454→    for (let g = 0; g < 5; g++) { // Lun-Ven
   455→      const day = new Date(currentMonday);
   456→      day.setDate(day.getDate() + g);
   457→      const dk = dateKey(day);
   458→      const meseKey = day.getFullYear() + '-' + String(day.getMonth()+1).padStart(2,'0');
   459→      
   460→      assegnazioni[dk] = {};
   461→      const personeRiposoOggi = [];
   462→      
   463→      // Determina la fascia per ogni gruppo questa settimana
   464→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   465→        if (gid === 'J') {
   466→          // Jolly: fisso 16:30-18:00 + sostituzioni
   467→          assegnazioni[dk]['Jolly'] = {
   468→            turno: 'fisso', entrata: '16:30', uscita: '18:00',
   469→            ore: 1.5, stato: 'lavoro', gruppo: 'J'
   470→          };
   471→          return;
   472→        }
   473→        
   474→        const fascia = getFasciaForWeek(weekIndex, gid);
   475→        const turniFascia = FASCE_TURNI[fascia]; // 2 ingressi
   476→        const numPersone = gruppo.persone.length;
   477→        const numIngressi = turniFascia.length; // sempre 2
   478→        
   479→        // Il riposo ruota SOLO nei gruppi che hanno più persone che ingressi
   480→        // Es: gruppo B (3 persone, 2 ingressi) → 1 persona a riposo che ruota
   481→        // Es: gruppo A (2 persone, 2 ingressi) → nessun riposo, tutti lavorano
   482→        const haRiposo = numPersone > numIngressi;
   483→        
   484→        if (haRiposo) {
   485→          // IL RIPOSO RUOTA: ogni giorno tocca a una persona diversa
   486→          const riposoIdx = giornoLavorativoIdx % numPersone;
   487→          
   488→          let turnoAssegnato = 0;
   489→          gruppo.persone.forEach((persona, idx) => {
   490→            if (idx === riposoIdx) {
   491→              assegnazioni[dk][persona] = {
   492→                turno: 'riposo', entrata: null, uscita: null,
   493→                ore: 0, stato: 'riposo', gruppo: gid
   494→              };
   495→              personeRiposoOggi.push(persona);
   496→            } else {
   497→              const entrata = turniFascia[turnoAssegnato];
   498→              const uscita = calcolaUscita(entrata);
   499→              assegnazioni[dk][persona] = {
   500→                turno: entrata, entrata: entrata, uscita: uscita,
   501→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   502→              };
   503→              turnoAssegnato++;
   504→            }
   505→          });
   506→        } else {
   507→          // Gruppo con stesso numero di persone e ingressi → tutti lavorano
   508→          gruppo.persone.forEach((persona, idx) => {
   509→            const entrata = turniFascia[idx];
   510→            const uscita = calcolaUscita(entrata);
   511→            assegnazioni[dk][persona] = {
   512→              turno: entrata, entrata: entrata, uscita: uscita,
   513→              ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   514→            };
   515→          });
   516→        }
   517→      });
   518→      
   519→      // 1 delle persone a riposo diventa reperibile
   520→      if (personeRiposoOggi.length >= 2) {
   521→        // Il secondo riposo diventa reperibile
   522→        assegnazioni[dk][personeRiposoOggi[1]] = {
   523→          ...assegnazioni[dk][personeRiposoOggi[1]],
   524→          stato: 'reperibile',
   525→          turno: 'reperibile'
   526→        };
   527→      }
   528→      
   529→      // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   530→      if (gruppi.J && personeRiposoOggi.length > 0) {
   531→        const sostituzioni = [];
   532→        personeRiposoOggi.forEach(p => {
   533→          const gid = getGruppoPersona(p);
   534→          if (gid && gid !== 'J') {
   535→            const fascia = getFasciaForWeek(weekIndex, gid);
   536→            const turniFascia = FASCE_TURNI[fascia];
   537→            // Trova quale turno avrebbe fatto (in base alla posizione nel gruppo senza il riposo)
   538→            const pIdx = gruppi[gid].persone.indexOf(p);
   539→            let turnoIdx = 0;
   540→            for (let i = 0; i < pIdx; i++) {
   541→              if (i !== (giornoLavorativoIdx % gruppi[gid].persone.length)) turnoIdx++;
   542→            }
   543→            if (turnoIdx < turniFascia.length) {
   544→              sostituzioni.push({
   545→                persona: p,
   546→                entrata: turniFascia[turnoIdx],
   547→                uscita: calcolaUscita(turniFascia[turnoIdx])
   548→              });
   549→            }
   550→          }
   551→        });
   552→        assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   553→        if (sostituzioni.length > 0) {
   554→          assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   555→        }
   556→      }
   557→      
   558→      giornoLavorativoIdx++;
   559→    }
   560→    
   561→    weekIndex++;
   562→    currentMonday.setDate(currentMonday.getDate() + 7);
   563→  }
   564→  
   565→  stato.assegnazioni = assegnazioni;
   566→  salvaStato();
   567→}

   606→function renderCalendario() {
   607→  const year = meseCorrente.getFullYear();
   608→  const month = meseCorrente.getMonth();
   609→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   610→  
   611→  const firstDay = new Date(year, month, 1);
   612→  const lastDay = new Date(year, month + 1, 0);
   613→  let startDow = firstDay.getDay() - 1;
   614→  if (startDow < 0) startDow = 6;
   615→  const today = new Date();
   616→  const gruppi = getGruppi();
   617→  
   618→  let html = '<table class="cal-table"><thead><tr>';
   619→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   620→  html += '</tr></thead><tbody>';
   621→  
   622→  let dayNum = 1, started = false;
   623→  for (let row = 0; row < 6; row++) {
   624→    if (dayNum > lastDay.getDate()) break;
   625→    html += '<tr>';
   626→    for (let col = 0; col < 7; col++) {
   627→      if (!started && col < startDow) { html += '<td></td>'; }
   628→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   629→      else {
   630→        started = true;
   631→        const d = new Date(year, month, dayNum);
   632→        const dk = dateKey(d);
   633→        const isToday = isSameDay(d, today);
   634→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;
   635→        
   636→        let cls = '';
   637→        if (isToday) cls += ' today';
   638→        if (isWeekend) cls += ' weekend';
   639→        
   640→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   641→        html += `<div class="day-num">${dayNum}</div>`;
   642→        
   643→        if (!isWeekend) {
   644→          const ass = stato.assegnazioni[dk] || {};
   645→          const hasData = Object.keys(ass).length > 0;
   646→          
   647→          if (hasData) {
   648→            // Mostra ogni persona con il proprio turno
   649→            Object.entries(gruppi).forEach(([gid, gruppo], grpIdx) => {
   650→              if (grpIdx > 0) html += '<div class="cal-grp-sep"></div>';
   651→              gruppo.persone.forEach(persona => {
   652→                const info = ass[persona];
   653→                if (!info) return;
   654→                
   655→                const dotColor = GRUPPO_COLORS[gid] || '#94a3b8';
   656→                html += '<div class="cal-person">';
   657→                html += `<span class="cal-person-dot" style="background:${dotColor}"></span>`;
   658→                
   659→                // Nome abbreviato (solo numero per A1→1, B2→2, Jolly→J)
   660→                const shortName = persona === 'Jolly' ? 'J' : persona.replace(/^[ABC]/, '');
   661→                html += `<span class="cal-person-name" style="color:${dotColor}">${shortName}</span>`;
   662→                
   663→                if (info.stato === 'lavoro' && info.entrata) {
   664→                  const tc = TURNI_COLORS[info.entrata] || TURNI_COLORS.fisso;
   665→                  html += `<span class="cal-person-time" style="background:${tc.bg};color:${tc.tx}">${info.entrata}</span>`;
   666→                } else if (info.stato === 'riposo') {
   667→                  html += `<span class="cal-person-riposo">Rip</span>`;
   668→                } else if (info.stato === 'reperibile') {
   669→                  html += `<span class="cal-person-reperib">Rep</span>`;
   670→                }
   671→                
   672→                html += '</div>';
   673→              });
   674→            });
   675→          } else {
   676→            // Nessun dato per questo giorno
   677→            html += '<div style="font-size:8px;color:#94a3b8;padding:4px 0">—</div>';
   678→          }
   679→        }
   680→        
   681→        html += '</td>';
   682→        dayNum++;
   683→      }
   684→    }
   685→    html += '</tr>';
   686→  }
   687→  html += '</tbody></table>';
   688→  document.getElementById('calendarioGrid').innerHTML = html;

   882→function renderOre() {
   883→  const year = meseCorrente.getFullYear();
   884→  const month = meseCorrente.getMonth();
   885→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   886→  
   887→  const firstDay = new Date(year, month, 1);
   888→  const lastDay = new Date(year, month + 1, 0);
   889→  const gruppi = getGruppi();
   890→  
   891→  let html = '';
   892→  
   893→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   894→    gruppo.persone.forEach(persona => {
   895→      let oreLavorate = 0, giorniLavorati = 0, giorniRiposo = 0;
   896→      
   897→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   898→        const dk = dateKey(d);
   899→        const dow = d.getDay();
   900→        if (dow === 0 || dow === 6) continue;
   901→        
   902→        const ass = stato.assegnazioni[dk];
   903→        if (!ass || !ass[persona]) continue;
   904→        
   905→        const info = ass[persona];
   906→        if (info.stato === 'lavoro') {
   907→          oreLavorate += info.ore || stato.oreTurno;
   908→          giorniLavorati++;
   909→        } else if (info.stato === 'riposo' || info.stato === 'reperibile') {
   910→          giorniRiposo++;
   911→        }
   912→      }
   913→      
   914→      if (gid === 'J') {
   915→        html += `<div class="ore-row">`;
   916→        html += `<span class="gruppo-badge gruppo-J" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   917→        html += `<div style="flex:1">`;
   918→        html += `<div style="font-size:12px">Fisso + sostituzioni: <strong>${oreLavorate.toFixed(1)}h</strong> (${giorniLavorati}gg)</div>`;
   919→        html += `</div></div>`;
   920→      } else {
   921→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
   922→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
   923→        const diff = oreLavorate - stato.oreTarget;
   924→        
   925→        html += `<div class="ore-row">`;
   926→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   927→        html += `<div style="flex:1">`;
   928→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
   929→        html += `<span>${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
   930→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
   931→        html += `</div>`;
   932→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
   933→        html += `</div></div>`;
   934→      }
   935→    });
   936→  });
   937→  
   938→  document.getElementById('oreList').innerHTML = html;
   939→}

   944→function exportExcel() {
   945→  const year = meseCorrente.getFullYear();
   946→  const month = meseCorrente.getMonth();
   947→  const firstDay = new Date(year, month, 1);
   948→  const lastDay = new Date(year, month + 1, 0);
   949→  const gruppi = getGruppi();
   950→  
   951→  // Intestazione
   952→  let csv = 'Data,Giorno,';
   953→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   954→    gruppo.persone.forEach(p => {
   955→      csv += p + ' Turno,' + p + ' Orario,' + p + ' Ore,';
   956→    });
   957→  });
   958→  csv += 'Presenti 9:30-15:30,\n';
   959→  
   960→  let totali = {};
   961→  Object.values(gruppi).forEach(g => g.persone.forEach(p => totali[p] = 0));
   962→  
   963→  for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   964→    const dk = dateKey(d);
   965→    const dow = d.getDay();
   966→    if (dow === 0 || dow === 6) continue;
   967→    
   968→    const dayName = GIORNI_FULL[dow === 0 ? 6 : dow - 1];
   969→    const ass = stato.assegnazioni[dk] || {};
   970→    
   971→    csv += `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()},${dayName},`;
   972→    
   973→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   974→      gruppo.persone.forEach(persona => {
   975→        const info = ass[persona];
   976→        if (!info) { csv += '—,,0,'; return; }
   977→        
   978→        if (info.stato === 'lavoro') {
   979→          csv += `${info.entrata},${info.entrata}-${info.uscita},${info.ore},`;
   980→          totali[persona] += info.ore || stato.oreTurno;
   981→        } else if (info.stato === 'reperibile') {
   982→          csv += 'Reperibile,,0,';
   983→        } else {
   984→          csv += 'Riposo,,0,';
   985→        }
   986→      });
   987→    });
   988→    
   989→    // Conta presenti alle 9:30
   990→    let presenti930 = 0;
   991→    Object.values(ass).forEach(info => {
   992→      if (info.stato === 'lavoro' && info.entrata && info.uscita) {
   993→        const e = tempoAMinuti(info.entrata);
   994→        const u = tempoAMinuti(info.uscita);
   995→        if (570 >= e && 930 < u) presenti930++;
   996→      }
   997→    });
   998→    csv += presenti930 + ',\n';
   999→  }
  1000→  
  1001→  // Totali
  1002→  csv += '\nTOTALI ORE,,';
  1003→  Object.values(gruppi).forEach(g => {
  1004→    g.persone.forEach(p => {
  1005→      csv += ',,' + totali[p].toFixed(1) + ',';
  1006→    });
  1007→  });
  1008→  csv += '\n';
  1009→  
  1010→  csv += 'TARGET,,';
  1011→  Object.values(gruppi).forEach(g => {
  1012→    g.persone.forEach(p => {
  1013→      csv += ',,' + (g.id === 'J' ? '—' : stato.oreTarget) + ',';
  1014→    });
  1015→  });
  1016→  
  1017→  const blob = new Blob(['\ufeff' + csv], {type: 'text/csv;charset=utf-8;'});
  1018→  const url = URL.createObjectURL(blob);
  1019→  const a = document.createElement('a');
  1020→  a.href = url;
  1021→  a.download = `Turnazioni_${MESI_IT[month]}_${year}.csv`;
  1022→  a.click();
  1023→  URL.revokeObjectURL(url);
  1024→}

   353→function getMonday(d) {
   354→  d = new Date(d);
   355→  const day = d.getDay();
   356→  const diff = d.getDate() - day + (day === 0 ? -6 : 1);
   357→  d.setDate(diff);
   358→  d.setHours(0,0,0,0);
   359→  return d;

   435→function generaAssegnazioni() {
   436→  const assegnazioni = {};
   437→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   438→  const today = new Date();
   439→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   440→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   441→  
   442→  const gruppi = getGruppi();
   443→  const isVar9 = stato.variante === 2;
   444→  
   445→  // Variante 2: lista piatta di tutte le 9 persone per la rotazione globale
   446→  let rotazioneGlobale = [];
   447→  if (isVar9) {
   448→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   449→      if (gid === 'J') return;
   450→      gruppo.persone.forEach(p => rotazioneGlobale.push({persona: p, gruppo: gid}));
   451→    });
   452→    // rotazioneGlobale = [{persona:'A1',gruppo:'A'}, {persona:'A2',gruppo:'A'}, {persona:'A3',gruppo:'A'},
   453→    //                      {persona:'B1',gruppo:'B'}, {persona:'B2',gruppo:'B'}, {persona:'B3',gruppo:'B'},
   454→    //                      {persona:'C1',gruppo:'C'}, {persona:'C2',gruppo:'C'}, {persona:'C3',gruppo:'C'}]
   455→  }
   456→  
   457→  let currentMonday = getMonday(startMonth);
   458→  const startMonday = getMonday(start);
   459→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   460→  let weekIndex = weeksDiffFromStart;
   461→  
   462→  while (currentMonday <= endMonth) {
   463→    for (let g = 0; g < 5; g++) { // Lun-Ven
   464→      const day = new Date(currentMonday);
   465→      day.setDate(day.getDate() + g);
   466→      const dk = dateKey(day);
   467→      
   468→      assegnazioni[dk] = {};
   469→      const personeRiposoOggi = [];
   470→      
   471→      // Indice globale del giorno lavorativo (deterministico, non dipende da quando si genera)
   472→      const globalDayIdx = weekIndex * 5 + g;
   473→      
   474→      if (isVar9) {
   475→        // ================================================================
   476→        // VARIANTE 2: 10 persone (3+3+3 + Jolly)
   477→        // 1 persona a riposo al giorno tra i 9, rotazione globale:
   478→        // A1 → A2 → A3 → B1 → B2 → B3 → C1 → C2 → C3 → repeat
   479→        // Chi riposa, riposa e basta
   480→        // Il Jolly (10ª persona): fisso 16:30-18:00 + copre il turno di chi riposa
   481→        // ================================================================
   482→        const riposoIdx = ((globalDayIdx % 9) + 9) % 9;
   483→        const chiRiposa = rotazioneGlobale[riposoIdx];
   484→        
   485→        // Trova la fascia del gruppo dove manca la persona, per il Jolly
   486→        const fasciaRiposo = getFasciaForWeek(weekIndex, chiRiposa.gruppo);
   487→        const turniFasciaRiposo = FASCE_TURNI[fasciaRiposo];
   488→        // Il Jolly copre uno dei 2 ingressi del gruppo dove manca la persona
   489→        const jollyIngresso = turniFasciaRiposo[0]; // primo ingresso
   490→        
   491→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   492→          if (gid === 'J') {
   493→            // Jolly: fisso 16:30-18:00 + copertura del turno di chi riposa
   494→            assegnazioni[dk]['Jolly'] = {
   495→              turno: 'jolly', entrata: jollyIngresso, uscita: calcolaUscita(jollyIngresso),
   496→              ore: stato.oreTurno, stato: 'lavoro', gruppo: 'J',
   497→              fisso: '16:30-18:00',
   498→              copre: chiRiposa.persona,
   499→              copreGruppo: chiRiposa.gruppo,
   500→              oreTotali: stato.oreTurno + 1.5 // turno + fisso serale
   501→            };
   502→            return;
   503→          }
   504→          
   505→          const fascia = getFasciaForWeek(weekIndex, gid);
   506→          const turniFascia = FASCE_TURNI[fascia]; // 2 ingressi
   507→          
   508→          let workingCount = 0;
   509→          gruppo.persone.forEach((persona) => {
   510→            if (persona === chiRiposa.persona && gid === chiRiposa.gruppo) {
   511→              // Questa persona riposa oggi
   512→              assegnazioni[dk][persona] = {
   513→                turno: 'riposo', entrata: null, uscita: null,
   514→                ore: 0, stato: 'riposo', gruppo: gid
   515→              };
   516→              personeRiposoOggi.push(persona);
   517→            } else {
   518→              // Lavora: assegna ingresso dalla fascia
   519→              const entrataIdx = workingCount % turniFascia.length;
   520→              const entrata = turniFascia[entrataIdx];
   521→              const uscita = calcolaUscita(entrata);
   522→              assegnazioni[dk][persona] = {
   523→                turno: entrata, entrata: entrata, uscita: uscita,
   524→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   525→              };
   526→              workingCount++;
   527→            }
   528→          });
   529→        });
   530→        
   531→      } else {
   532→        // ================================================================
   533→        // VARIANTE 1: 8 persone (2+3+3) + Jolly
   534→        // Il riposo ruota dentro ogni gruppo che ha piu persone che ingressi
   535→        // ================================================================
   536→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   537→          if (gid === 'J') {
   538→            // Jolly: fisso 16:30-18:00 + sostituzioni
   539→            assegnazioni[dk]['Jolly'] = {
   540→              turno: 'fisso', entrata: '16:30', uscita: '18:00',
   541→              ore: 1.5, stato: 'lavoro', gruppo: 'J'
   542→            };
   543→            return;
   544→          }
   545→          
   546→          const fascia = getFasciaForWeek(weekIndex, gid);
   547→          const turniFascia = FASCE_TURNI[fascia];
   548→          const numPersone = gruppo.persone.length;
   549→          const numIngressi = turniFascia.length;
   550→          const haRiposo = numPersone > numIngressi;
   551→          
   552→          if (haRiposo) {
   553→            const riposoIdx = ((globalDayIdx % numPersone) + numPersone) % numPersone;
   554→            let turnoAssegnato = 0;
   555→            gruppo.persone.forEach((persona, idx) => {
   556→              if (idx === riposoIdx) {
   557→                assegnazioni[dk][persona] = {
   558→                  turno: 'riposo', entrata: null, uscita: null,
   559→                  ore: 0, stato: 'riposo', gruppo: gid
   560→                };
   561→                personeRiposoOggi.push(persona);
   562→              } else {
   563→                const entrata = turniFascia[turnoAssegnato];
   564→                const uscita = calcolaUscita(entrata);
   565→                assegnazioni[dk][persona] = {
   566→                  turno: entrata, entrata: entrata, uscita: uscita,
   567→                  ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   568→                };
   569→                turnoAssegnato++;
   570→              }
   571→            });
   572→          } else {
   573→            gruppo.persone.forEach((persona, idx) => {
   574→              const entrata = turniFascia[idx];
   575→              const uscita = calcolaUscita(entrata);
   576→              assegnazioni[dk][persona] = {
   577→                turno: entrata, entrata: entrata, uscita: uscita,
   578→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   579→              };
   580→            });
   581→          }
   582→        });
   583→        
   584→        // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   585→        if (gruppi.J && personeRiposoOggi.length > 0) {
   586→          const sostituzioni = [];
   587→          personeRiposoOggi.forEach(p => {
   588→            const gid = getGruppoPersona(p);
   589→            if (gid && gid !== 'J') {
   590→              const fascia = getFasciaForWeek(weekIndex, gid);
   591→              const turniFascia = FASCE_TURNI[fascia];
   592→              const pIdx = gruppi[gid].persone.indexOf(p);
   593→              let turnoIdx = 0;
   594→              const riposoIdxGruppo = ((globalDayIdx % gruppi[gid].persone.length) + gruppi[gid].persone.length) % gruppi[gid].persone.length;
   595→              for (let i = 0; i < pIdx; i++) {
   596→                if (i !== riposoIdxGruppo) turnoIdx++;
   597→              }
   598→              if (turnoIdx < turniFascia.length) {
   599→                sostituzioni.push({
   600→                  persona: p,
   601→                  entrata: turniFascia[turnoIdx],
   602→                  uscita: calcolaUscita(turniFascia[turnoIdx])
   603→                });
   604→              }
   605→            }
   606→          });
   607→          assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   608→          if (sostituzioni.length > 0) {
   609→            assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   610→          }
   611→        }
   612→      }
   613→    }
   614→    
   615→    weekIndex++;
   616→    currentMonday.setDate(currentMonday.getDate() + 7);
   617→  }
   618→  
   619→  stato.assegnazioni = assegnazioni;
   620→  salvaStato();
   621→}

   672→function renderCalendario() {
   673→  const year = meseCorrente.getFullYear();
   674→  const month = meseCorrente.getMonth();
   675→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   676→  
   677→  const firstDay = new Date(year, month, 1);
   678→  const lastDay = new Date(year, month + 1, 0);
   679→  let startDow = firstDay.getDay() - 1;
   680→  if (startDow < 0) startDow = 6;
   681→  const today = new Date();
   682→  const gruppi = getGruppi();
   683→  
   684→  let html = '<table class="cal-table"><thead><tr>';
   685→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   686→  html += '</tr></thead><tbody>';
   687→  
   688→  let dayNum = 1, started = false;
   689→  for (let row = 0; row < 6; row++) {
   690→    if (dayNum > lastDay.getDate()) break;
   691→    html += '<tr>';
   692→    for (let col = 0; col < 7; col++) {
   693→      if (!started && col < startDow) { html += '<td></td>'; }
   694→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   695→      else {
   696→        started = true;
   697→        const d = new Date(year, month, dayNum);
   698→        const dk = dateKey(d);
   699→        const isToday = isSameDay(d, today);
   700→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;
   701→        
   702→        let cls = '';
   703→        if (isToday) cls += ' today';
   704→        if (isWeekend) cls += ' weekend';
   705→        
   706→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   707→        html += `<div class="day-num">${dayNum}</div>`;
   708→        
   709→        if (!isWeekend) {
   710→          const ass = stato.assegnazioni[dk] || {};
   711→          const hasData = Object.keys(ass).length > 0;
   712→          
   713→          if (hasData) {

   811→    });
   812→    html += `</div>`;
   813→  });

   816→  html += `<div style="margin-top:12px;padding:10px;background:#f0fdf4;border-radius:8px;font-size:12px">`;
   817→  html += `<strong>Copertura oraria:</strong><br>`;

   820→  fasce.forEach(fascia => {

   822→    Object.values(ass).forEach(info => {
   823→      if (info.stato === 'lavoro' && info.entrata && info.uscita) {
   824→        const e = tempoAMinuti(info.entrata);

   831→        const f = tempoAMinuti(fascia);

   848→function tempoAMinuti(t) {
   849→  if (!t) return 0;
   850→  const [h,m] = t.split(':').map(Number);
   851→  return h * 60 + (m || 0);

   940→    '13:00':6,'14:00':6,'15:00':6,'15:30':6,'16:00':5,'16:30':4,'17:00':3,'17:30':2,'18:00':1

   962→    html += `</div>`;

   968→// ORE
   969→function renderOre() {
   970→  const year = meseCorrente.getFullYear();
   971→  const month = meseCorrente.getMonth();
   972→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   973→  
   974→  const firstDay = new Date(year, month, 1);
   975→  const lastDay = new Date(year, month + 1, 0);
   976→  const gruppi = getGruppi();
   977→  
   978→  let html = '';
   979→  
   980→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   981→    gruppo.persone.forEach(persona => {
   982→      let oreLavorate = 0, giorniLavorati = 0, giorniRiposo = 0;
   983→      
   984→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   985→        const dk = dateKey(d);
   986→        const dow = d.getDay();
   987→        if (dow === 0 || dow === 6) continue;
   988→        
   989→        const ass = stato.assegnazioni[dk];
   990→        if (!ass || !ass[persona]) continue;
   991→        
   992→        const info = ass[persona];
   993→        if (info.stato === 'lavoro') {
   994→          oreLavorate += info.ore || stato.oreTurno;
   995→          giorniLavorati++;
   996→        } else if (info.stato === 'riposo') {
   997→          giorniRiposo++;
   998→        }
   999→      }
  1000→      
  1001→      if (gid === 'J') {
  1002→        // Calcola le ore del Jolly includendo il fisso serale
  1003→        let oreJolly = 0;
  1004→        for (let d2 = new Date(firstDay); d2 <= lastDay; d2.setDate(d2.getDate() + 1)) {
  1005→          const dk2 = dateKey(d2);
  1006→          const dow2 = d2.getDay();
  1007→          if (dow2 === 0 || dow2 === 6) continue;
  1008→          const assJ = stato.assegnazioni[dk2];
  1009→          if (assJ && assJ['Jolly']) {
  1010→            const jInfo = assJ['Jolly'];
  1011→            if (jInfo.oreTotali) oreJolly += jInfo.oreTotali;
  1012→            else oreJolly += (jInfo.ore || stato.oreTurno) + 1.5;
  1013→          }
  1014→        }
  1015→        const jollyDiff = oreJolly - stato.oreTarget;
  1016→        const jollyPct = Math.min(100, Math.round((oreJolly / stato.oreTarget) * 100));
  1017→        const jollyBarClass = jollyPct >= 95 ? 'ok' : jollyPct >= 80 ? 'warn' : 'danger';
  1018→        html += `<div class="ore-row">`;
  1019→        html += `<span class="gruppo-badge gruppo-J" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
  1020→        html += `<div style="flex:1">`;
  1021→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
  1022→        html += `<span>Turno + 16:30-18:00</span>`;
  1023→        html += `<span style="font-weight:700;color:${jollyBarClass==='ok'?'var(--success)':jollyBarClass==='warn'?'var(--accent)':'var(--danger)'}">${oreJolly.toFixed(1)}h ${jollyDiff>=0?'✓':'('+jollyDiff.toFixed(1)+'h)'}</span>`;
  1024→        html += `</div>`;
  1025→        html += `<div class="ore-progress"><div class="ore-bar ${jollyBarClass}" style="width:${jollyPct}%"></div></div>`;
  1026→        html += `</div></div>`;
  1027→      } else {
  1028→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
  1029→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
  1030→        const diff = oreLavorate - stato.oreTarget;
  1031→        
  1032→        html += `<div class="ore-row">`;
  1033→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
  1034→        html += `<div style="flex:1">`;
  1035→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
  1036→        html += `<span>${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
  1037→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
  1038→        html += `</div>`;
  1039→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
  1040→        html += `</div></div>`;
  1041→      }
  1042→    });
  1043→  });
  1044→  
  1045→  document.getElementById('oreList').innerHTML = html;
  1046→}

  1070→  for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
  1071→    const dk = dateKey(d);

  1081→      gruppo.persone.forEach(persona => {

  1088→
  1089→        } else {
  1090→          csv += 'Riposo,,0,';

   550→   550→function generaAssegnazioni() {
   551→   551→  const assegnazioni = {};
   552→   552→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   553→   553→  const today = new Date();
   554→   554→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   555→   555→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   556→   556→  
   557→   557→  const gruppi = getGruppi();
   558→   558→  const isVar9 = stato.variante === 2;
   559→   559→  
   560→   560→  // Variante 2: lista piatta di tutte le 9 persone per la rotazione globale
   561→   561→  let rotazioneGlobale = [];
   562→   562→  if (isVar9) {
   563→   563→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   564→   564→      if (gid === 'J') return;
   565→   565→      gruppo.persone.forEach(p => rotazioneGlobale.push({persona: p, gruppo: gid}));
   566→   566→    });
   567→   567→  }
   568→   568→  
   569→   569→  let currentMonday = getMonday(startMonth);
   570→   570→  const startMonday = getMonday(start);
   571→   571→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   572→   572→  let weekIndex = weeksDiffFromStart;
   573→   573→  
   574→   574→  while (currentMonday <= endMonth) {
   575→   575→    for (let g = 0; g < 5; g++) { // Lun-Ven
   576→   576→      const day = new Date(currentMonday);
   577→   577→      day.setDate(day.getDate() + g);
   578→   578→      const dk = dateKey(day);
   579→   579→      
   580→   580→      assegnazioni[dk] = {};
   581→   581→      const personeRiposoOggi = [];
   582→   582→      
   583→   583→      const globalDayIdx = weekIndex * 5 + g;
   584→   584→      
   585→   585→      if (isVar9) {
   586→   586→        const riposoIdx = ((globalDayIdx % 9) + 9) % 9;
   587→   587→        const chiRiposa = rotazioneGlobale[riposoIdx];
   588→   588→        
   589→   589→        const fasciaRiposo = getFasciaForWeek(weekIndex, chiRiposa.gruppo);
   590→   590→        const turniFasciaRiposo = FASCE_TURNI[fasciaRiposo];
   591→   591→        const jollyIngresso = turniFasciaRiposo[0];
   592→   592→        
   593→   593→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   594→   594→          if (gid === 'J') {
   595→   595→            assegnazioni[dk]['Jolly'] = {
   596→   596→              turno: 'jolly', entrata: jollyIngresso, uscita: calcolaUscita(jollyIngresso),
   597→   597→              ore: stato.oreTurno, stato: 'lavoro', gruppo: 'J',
   598→   598→              fisso: '16:30-18:00',
   599→   599→              copre: chiRiposa.persona,
   600→   600→              copreGruppo: chiRiposa.gruppo,
   601→   601→              oreTotali: stato.oreTurno + 1.5
   602→   602→            };
   603→   603→            return;
   604→   604→          }

   606→   606→          const fascia = getFasciaForWeek(weekIndex, gid);

   648→   648→            let turnoAssegnato = 0;

   681→   681→          personeRiposoOggi.forEach(p => {

   686→   686→              const pIdx = gruppi[gid].persone.indexOf(p);

   688→   688→              const riposoIdxGruppo = ((globalDayIdx % gruppi[gid].persone.length) + gruppi[gid].persone.length) % gruppi[gid].persone.length;

   992→   992→// Save override for a specific person on a specific day
   993→   993→function salvaOverride(dk, persona) {
   994→   994→  const eInput = document.getElementById('edit_e_' + dk + '_' + persona);
   995→   995→  const uInput = document.getElementById('edit_u_' + dk + '_' + persona);
   996→   996→  if (!eInput || !uInput) return;
   997→   997→  
   998→   998→  const entrata = eInput.value.trim();
   999→   999→  const uscita = uInput.value.trim();
  1000→  1000→  
  1001→  1001→  if (!entrata || !uscita) {
  1002→  1002→    alert('Inserisci sia entrata che uscita.');
  1003→  1003→    return;
  1004→  1004→  }
  1005→  1005→  
  1006→  1006→  // Validate time format
  1007→  1007→  const timeRegex = /^\d{1,2}:\d{2}$/;
  1008→  1008→  if (!timeRegex.test(entrata) || !timeRegex.test(uscita)) {
  1009→  1009→    alert('Formato orario non valido. Usa HH:MM (es. 8:00, 15:30).');
  1010→  1010→    return;
  1011→  1011→  }
  1012→  1012→  
  1013→  1013→  if (!stato.orariOverride) stato.orariOverride = {};
  1014→  1014→  if (!stato.orariOverride[dk]) stato.orariOverride[dk] = {};
  1015→  1015→  
  1016→  1016→  // Check if the override is same as original → remove it
  1017→  1017→  const baseInfo = stato.assegnazioni[dk] && stato.assegnazioni[dk][persona];
  1018→  1018→  if (baseInfo && baseInfo.entrata === entrata && baseInfo.uscita === uscita) {
  1019→  1019→    delete stato.orariOverride[dk][persona];
  1020→  1020→    if (Object.keys(stato.orariOverride[dk]).length === 0) {
  1021→  1021→      delete stato.orariOverride[dk];
  1022→  1022→    }
  1023→  1023→  } else {
  1024→  1024→    stato.orariOverride[dk][persona] = { entrata, uscita };
  1025→  1025→  }
  1026→  1026→  
  1027→  1027→  salvaStato();
  1028→  1028→  // Re-render the modal with updated data
  1029→  1029→  mostraGiorno(dk);
  1030→  1030→  renderCalendario();
  1031→  1031→}

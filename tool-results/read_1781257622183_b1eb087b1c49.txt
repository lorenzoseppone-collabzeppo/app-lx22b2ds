   435→   435→function generaAssegnazioni() {
   436→   436→  const assegnazioni = {};
   437→   437→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   438→   438→  const today = new Date();
   439→   439→  const startMonth = new Date(today.getFullYear(), today.getMonth() - 1, 1);
   440→   440→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   441→   441→  
   442→   442→  const gruppi = getGruppi();
   443→   443→  const isVar9 = stato.variante === 2;
   444→   444→  
   445→   445→  // Variante 2: lista piatta di tutte le 9 persone per la rotazione globale
   446→   446→  let rotazioneGlobale = [];
   447→   447→  if (isVar9) {
   448→   448→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   449→   449→      if (gid === 'J') return;
   450→   450→      gruppo.persone.forEach(p => rotazioneGlobale.push({persona: p, gruppo: gid}));
   451→   451→    });
   452→   452→    // rotazioneGlobale = [{persona:'A1',gruppo:'A'}, {persona:'A2',gruppo:'A'}, {persona:'A3',gruppo:'A'},
   453→   453→    //                      {persona:'B1',gruppo:'B'}, {persona:'B2',gruppo:'B'}, {persona:'B3',gruppo:'B'},
   454→   454→    //                      {persona:'C1',gruppo:'C'}, {persona:'C2',gruppo:'C'}, {persona:'C3',gruppo:'C'}]
   455→   455→  }
   456→   456→  
   457→   457→  let currentMonday = getMonday(startMonth);
   458→   458→  const startMonday = getMonday(start);
   459→   459→  const weeksDiffFromStart = Math.round((currentMonday - startMonday) / (7 * 86400000));
   460→   460→  let weekIndex = weeksDiffFromStart;
   461→   461→  
   462→   462→  while (currentMonday <= endMonth) {
   463→   463→    for (let g = 0; g < 5; g++) { // Lun-Ven
   464→   464→      const day = new Date(currentMonday);
   465→   465→      day.setDate(day.getDate() + g);
   466→   466→      const dk = dateKey(day);
   467→   467→      
   468→   468→      assegnazioni[dk] = {};
   469→   469→      const personeRiposoOggi = [];
   470→   470→      
   471→   471→      // Indice globale del giorno lavorativo (deterministico, non dipende da quando si genera)
   472→   472→      const globalDayIdx = weekIndex * 5 + g;
   473→   473→      
   474→   474→      if (isVar9) {
   475→   475→        // ================================================================
   476→   476→        // VARIANTE 2: 10 persone (3+3+3 + Jolly)
   477→   477→        // 1 persona a riposo al giorno tra i 9, rotazione globale:
   478→   478→        // A1 → A2 → A3 → B1 → B2 → B3 → C1 → C2 → C3 → repeat
   479→   479→        // Chi riposa, riposa e basta
   480→   480→        // Il Jolly (10ª persona): fisso 16:30-18:00 + copre il turno di chi riposa
   481→   481→        // ================================================================
   482→   482→        const riposoIdx = ((globalDayIdx % 9) + 9) % 9;
   483→   483→        const chiRiposa = rotazioneGlobale[riposoIdx];
   484→   484→        
   485→   485→        // Trova la fascia del gruppo dove manca la persona, per il Jolly
   486→   486→        const fasciaRiposo = getFasciaForWeek(weekIndex, chiRiposa.gruppo);
   487→   487→        const turniFasciaRiposo = FASCE_TURNI[fasciaRiposo];
   488→   488→        // Il Jolly copre uno dei 2 ingressi del gruppo dove manca la persona
   489→   489→        const jollyIngresso = turniFasciaRiposo[0]; // primo ingresso
   490→   490→        
   491→   491→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   492→   492→          if (gid === 'J') {
   493→   493→            // Jolly: fisso 16:30-18:00 + copertura del turno di chi riposa
   494→   494→            assegnazioni[dk]['Jolly'] = {
   495→   495→              turno: 'jolly', entrata: jollyIngresso, uscita: calcolaUscita(jollyIngresso),
   496→   496→              ore: stato.oreTurno, stato: 'lavoro', gruppo: 'J',
   497→   497→              fisso: '16:30-18:00',
   498→   498→              copre: chiRiposa.persona,
   499→   499→              copreGruppo: chiRiposa.gruppo,
   500→   500→              oreTotali: stato.oreTurno + 1.5 // turno + fisso serale
   501→   501→            };
   502→   502→            return;
   503→   503→          }
   504→   504→          
   505→   505→          const fascia = getFasciaForWeek(weekIndex, gid);
   506→   506→          const turniFascia = FASCE_TURNI[fascia]; // 2 ingressi
   507→   507→          
   508→   508→          let workingCount = 0;
   509→   509→          gruppo.persone.forEach((persona) => {
   510→   510→            if (persona === chiRiposa.persona && gid === chiRiposa.gruppo) {
   511→   511→              // Questa persona riposa oggi
   512→   512→              assegnazioni[dk][persona] = {
   513→   513→                turno: 'riposo', entrata: null, uscita: null,
   514→   514→                ore: 0, stato: 'riposo', gruppo: gid
   515→   515→              };
   516→   516→              personeRiposoOggi.push(persona);
   517→   517→            } else {
   518→   518→              // Lavora: assegna ingresso dalla fascia
   519→   519→              const entrataIdx = workingCount % turniFascia.length;
   520→   520→              const entrata = turniFascia[entrataIdx];
   521→   521→              const uscita = calcolaUscita(entrata);
   522→   522→              assegnazioni[dk][persona] = {
   523→   523→                turno: entrata, entrata: entrata, uscita: uscita,
   524→   524→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   525→   525→              };
   526→   526→              workingCount++;
   527→   527→            }
   528→   528→          });
   529→   529→        });
   530→   530→        
   531→   531→      } else {
   532→   532→        // ================================================================
   533→   533→        // VARIANTE 1: 8 persone (2+3+3) + Jolly
   534→   534→        // Il riposo ruota dentro ogni gruppo che ha piu persone che ingressi
   535→   535→        // ================================================================
   536→   536→        Object.entries(gruppi).forEach(([gid, gruppo]) => {
   537→   537→          if (gid === 'J') {
   538→   538→            // Jolly: fisso 16:30-18:00 + sostituzioni
   539→   539→            assegnazioni[dk]['Jolly'] = {
   540→   540→              turno: 'fisso', entrata: '16:30', uscita: '18:00',
   541→   541→              ore: 1.5, stato: 'lavoro', gruppo: 'J'
   542→   542→            };
   543→   543→            return;
   544→   544→          }
   545→   545→          
   546→   546→          const fascia = getFasciaForWeek(weekIndex, gid);
   547→   547→          const turniFascia = FASCE_TURNI[fascia];
   548→   548→          const numPersone = gruppo.persone.length;
   549→   549→          const numIngressi = turniFascia.length;
   550→   550→          const haRiposo = numPersone > numIngressi;
   551→   551→          
   552→   552→          if (haRiposo) {
   553→   553→            const riposoIdx = ((globalDayIdx % numPersone) + numPersone) % numPersone;
   554→   554→            let turnoAssegnato = 0;
   555→   555→            gruppo.persone.forEach((persona, idx) => {
   556→   556→              if (idx === riposoIdx) {
   557→   557→                assegnazioni[dk][persona] = {
   558→   558→                  turno: 'riposo', entrata: null, uscita: null,
   559→   559→                  ore: 0, stato: 'riposo', gruppo: gid
   560→   560→                };
   561→   561→                personeRiposoOggi.push(persona);
   562→   562→              } else {
   563→   563→                const entrata = turniFascia[turnoAssegnato];
   564→   564→                const uscita = calcolaUscita(entrata);
   565→   565→                assegnazioni[dk][persona] = {
   566→   566→                  turno: entrata, entrata: entrata, uscita: uscita,
   567→   567→                  ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   568→   568→                };
   569→   569→                turnoAssegnato++;
   570→   570→              }
   571→   571→            });
   572→   572→          } else {
   573→   573→            gruppo.persone.forEach((persona, idx) => {
   574→   574→              const entrata = turniFascia[idx];
   575→   575→              const uscita = calcolaUscita(entrata);
   576→   576→              assegnazioni[dk][persona] = {
   577→   577→                turno: entrata, entrata: entrata, uscita: uscita,
   578→   578→                ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   579→   579→              };
   580→   580→            });
   581→   581→          }
   582→   582→        });
   583→   583→        
   584→   584→        // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   585→   585→        if (gruppi.J && personeRiposoOggi.length > 0) {
   586→   586→          const sostituzioni = [];
   587→   587→          personeRiposoOggi.forEach(p => {
   588→   588→            const gid = getGruppoPersona(p);
   589→   589→            if (gid && gid !== 'J') {
   590→   590→              const fascia = getFasciaForWeek(weekIndex, gid);
   591→   591→              const turniFascia = FASCE_TURNI[fascia];
   592→   592→              const pIdx = gruppi[gid].persone.indexOf(p);
   593→   593→              let turnoIdx = 0;
   594→   594→              const riposoIdxGruppo = ((globalDayIdx % gruppi[gid].persone.length) + gruppi[gid].persone.length) % gruppi[gid].persone.length;
   595→   595→              for (let i = 0; i < pIdx; i++) {
   596→   596→                if (i !== riposoIdxGruppo) turnoIdx++;
   597→   597→              }
   598→   598→              if (turnoIdx < turniFascia.length) {
   599→   599→                sostituzioni.push({
   600→   600→                  persona: p,
   601→   601→                  entrata: turniFascia[turnoIdx],
   602→   602→                  uscita: calcolaUscita(turniFascia[turnoIdx])
   603→   603→                });
   604→   604→              }
   605→   605→            }
   606→   606→          });
   607→   607→          assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   608→   608→          if (sostituzioni.length > 0) {
   609→   609→            assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   610→   610→          }
   611→   611→        }
   612→   612→      }
   613→   613→    }
   614→   614→    
   615→   615→    weekIndex++;
   616→   616→    currentMonday.setDate(currentMonday.getDate() + 7);
   617→   617→  }
   618→   618→  
   619→   619→  stato.assegnazioni = assegnazioni;
   620→   620→  salvaStato();
   621→   621→}

   672→   672→function renderCalendario() {

   674→   674→  const month = meseCorrente.getMonth();
   675→   675→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   676→   676→  
   677→   677→  const firstDay = new Date(year, month, 1);
   678→   678→  const lastDay = new Date(year, month + 1, 0);
   679→   679→  let startDow = firstDay.getDay() - 1;
   680→   680→  if (startDow < 0) startDow = 6;
   681→   681→  const today = new Date();
   682→   682→  const gruppi = getGruppi();

   686→   686→  html += '</tr></thead><tbody>';

   688→   688→  let dayNum = 1, started = false;
   689→   689→  for (let row = 0; row < 6; row++) {
   690→   690→    if (dayNum > lastDay.getDate()) break;
   691→   691→    html += '<tr>';
   692→   692→    for (let col = 0; col < 7; col++) {
   693→   693→      if (!started && col < startDow) { html += '<td></td>'; }
   694→   694→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   695→   695→      else {
   696→   696→        started = true;
   697→   697→        const d = new Date(year, month, dayNum);
   698→   698→        const dk = dateKey(d);
   699→   699→        const isToday = isSameDay(d, today);
   700→   700→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;

   706→   706→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   707→   707→        html += `<div class="day-num">${dayNum}</div>`;

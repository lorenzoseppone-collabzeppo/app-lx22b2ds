   348→function getMonday(d) {
   349→  d = new Date(d);
   350→  const day = d.getDay();
   351→  const diff = d.getDate() - day + (day === 0 ? -6 : 1);
   352→  d.setDate(diff);
   353→  d.setHours(0,0,0,0);
   354→  return d;
   355→}

   357→function dateKey(d) {
   358→  const y = d.getFullYear();
   359→  const m = String(d.getMonth()+1).padStart(2,'0');
   360→  const dd = String(d.getDate()).padStart(2,'0');
   361→  return `${y}-${m}-${dd}`;
   362→}

   368→function getWeekNumber(d) {
   369→  const onejan = new Date(d.getFullYear(),0,1);
   370→  return Math.ceil((((d - onejan)/86400000)+onejan.getDay()+1)/7);
   371→}

   429→function generaAssegnazioni() {
   430→  const assegnazioni = {};
   431→  const start = new Date(stato.inizioRotazione + 'T00:00:00');
   432→  const today = new Date();
   433→  const endMonth = new Date(today.getFullYear(), today.getMonth() + 6, 0);
   434→  
   435→  const gruppi = getGruppi();
   436→  const tuttiNomi = [];
   437→  Object.values(gruppi).forEach(g => g.persone.forEach(p => tuttiNomi.push(p)));
   438→  
   439→  let currentMonday = new Date(start);
   440→  let weekIndex = 0;
   441→  
   442→  // Traccia riposi per equità mensile
   443→  const riposiMesePersona = {}; // meseKey -> persona -> count
   444→  
   445→  while (currentMonday <= endMonth) {
   446→    for (let g = 0; g < 5; g++) { // Lun-Ven
   447→      const day = new Date(currentMonday);
   448→      day.setDate(day.getDate() + g);
   449→      const dk = dateKey(day);
   450→      const meseKey = day.getFullYear() + '-' + String(day.getMonth()+1).padStart(2,'0');
   451→      
   452→      assegnazioni[dk] = {};
   453→      
   454→      // Determina la fascia per ogni gruppo questa settimana
   455→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   456→        if (gid === 'J') {
   457→          // Jolly: fisso 16:30-18:00 + sostituzioni
   458→          assegnazioni[dk]['Jolly'] = {
   459→            turno: 'fisso', entrata: '16:30', uscita: '18:00',
   460→            ore: 1.5, stato: 'lavoro', gruppo: 'J'
   461→          };
   462→          return;
   463→        }
   464→        
   465→        const fascia = getFasciaForWeek(weekIndex, gid);
   466→        const turniFascia = FASCE_TURNI[fascia];
   467→        
   468→        gruppo.persone.forEach((persona, idx) => {
   469→          if (idx < turniFascia.length) {
   470→            const entrata = turniFascia[idx];
   471→            const uscita = calcolaUscita(entrata);
   472→            assegnazioni[dk][persona] = {
   473→              turno: entrata, entrata: entrata, uscita: uscita,
   474→              ore: stato.oreTurno, stato: 'lavoro', gruppo: gid
   475→            };
   476→          } else {
   477→            // Terza persona → riposo o reperibile
   478→            assegnazioni[dk][persona] = {
   479→              turno: 'riposo', entrata: null, uscita: null,
   480→              ore: 0, stato: 'riposo', gruppo: gid
   481→            };
   482→          }
   483→        });
   484→      });
   485→      
   486→      // SISTEMA RIPOSI: 2 persone a riposo, da gruppi diversi
   487→      // 1 delle 2 è reperibile
   488→      const personeRiposo = [];
   489→      Object.entries(gruppi).forEach(([gid, gruppo]) => {
   490→        gruppo.persone.forEach(persona => {
   491→          if (assegnazioni[dk][persona] && assegnazioni[dk][persona].stato === 'riposo') {
   492→            personeRiposo.push(persona);
   493→          }
   494→        });
   495→      });
   496→      
   497→      // Se ci sono già 2+ persone a riposo, ok
   498→      // Se ce n'è solo 1, aggiungiamo 1 persona da un altro gruppo
   499→      if (personeRiposo.length < stato.giorniRiposo) {
   500→        // Trova chi può andare a riposo (da gruppi diversi da chi è già a riposo)
   501→        const gruppiGiaRiposo = new Set(personeRiposo.map(p => assegnazioni[dk][p].gruppo));
   502→        
   503→        // Scegli chi ha riposato di meno questo mese
   504→        const conteggio = {};
   505→        tuttiNomi.forEach(p => conteggio[p] = 0);
   506→        Object.keys(assegnazioni).forEach(rdk => {
   507→          if (rdk.startsWith(meseKey) && assegnazioni[rdk]) {
   508→            Object.entries(assegnazioni[rdk]).forEach(([persona, info]) => {
   509→              if (info.stato === 'riposo' && conteggio[persona] !== undefined) conteggio[persona]++;
   510→            });
   511→          }
   512→        });
   513→        
   514→        const candidati = tuttiNomi.filter(p => {
   515→          if (personeRiposo.includes(p)) return false;
   516→          if (p === 'Jolly') return false;
   517→          const grp = getGruppoPersona(p);
   518→          return !gruppiGiaRiposo.has(grp);
   519→        }).sort((a,b) => conteggio[a] - conteggio[b]);
   520→        
   521→        while (personeRiposo.length < stato.giorniRiposo && candidati.length > 0) {
   522→          const persona = candidati.shift();
   523→          const gruppoOld = assegnazioni[dk][persona];
   524→          const nuovoGruppoRiposo = getGruppoPersona(persona);
   525→          
   526→          // Metti a riposo
   527→          assegnazioni[dk][persona] = {
   528→            turno: 'riposo', entrata: null, uscita: null,
   529→            ore: 0, stato: 'riposo', gruppo: nuovoGruppoRiposo
   530→          };
   531→          personeRiposo.push(persona);
   532→          
   533→          // Aggiungi il gruppo ai già in riposo
   534→          gruppiGiaRiposo.add(nuovoGruppoRiposo);
   535→        }
   536→      }
   537→      
   538→      // Segna 1 riposo come reperibile
   539→      if (personeRiposo.length >= 2) {
   540→        // Il secondo diventa reperibile
   541→        assegnazioni[dk][personeRiposo[1]] = {
   542→          ...assegnazioni[dk][personeRiposo[1]],
   543→          stato: 'reperibile',
   544→          turno: 'reperibile'
   545→        };
   546→      }
   547→      
   548→      // Jolly: se ci sono persone a riposo, il Jolly copre i loro turni
   549→      if (gruppi.J && personeRiposo.length > 0) {
   550→        const sostituzioni = [];
   551→        personeRiposo.forEach(p => {
   552→          // Trova il turno che avrebbe dovuto fare
   553→          const gid = getGruppoPersona(p);
   554→          if (gid && gid !== 'J') {
   555→            const fascia = getFasciaForWeek(weekIndex, gid);
   556→            const turniFascia = FASCE_TURNI[fascia];
   557→            const idx = gruppi[gid].persone.indexOf(p);
   558→            if (idx >= 0 && idx < turniFascia.length) {
   559→              sostituzioni.push({
   560→                persona: p,
   561→                entrata: turniFascia[idx],
   562→                uscita: calcolaUscita(turniFascia[idx])
   563→              });
   564→            }
   565→          }
   566→        });
   567→        assegnazioni[dk]['Jolly'].sostituzioni = sostituzioni;
   568→        if (sostituzioni.length > 0) {
   569→          assegnazioni[dk]['Jolly'].oreTotali = 1.5 + sostituzioni.length * stato.oreTurno;
   570→        }
   571→      }
   572→    }
   573→    
   574→    weekIndex++;
   575→    currentMonday.setDate(currentMonday.getDate() + 7);
   576→  }
   577→  
   578→  stato.assegnazioni = assegnazioni;
   579→  salvaStato();
   580→}

   617→function renderCalendario() {
   618→  const year = meseCorrente.getFullYear();
   619→  const month = meseCorrente.getMonth();
   620→  document.getElementById('calMeseTitolo').textContent = MESI_IT[month] + ' ' + year;
   621→  
   622→  const firstDay = new Date(year, month, 1);
   623→  const lastDay = new Date(year, month + 1, 0);
   624→  let startDow = firstDay.getDay() - 1;
   625→  if (startDow < 0) startDow = 6;
   626→  const today = new Date();
   627→  
   628→  let html = '<table class="cal-table"><thead><tr>';
   629→  GIORNI_IT.forEach(g => html += `<th>${g}</th>`);
   630→  html += '</tr></thead><tbody>';
   631→  
   632→  let dayNum = 1, started = false;
   633→  for (let row = 0; row < 6; row++) {
   634→    if (dayNum > lastDay.getDate()) break;
   635→    html += '<tr>';
   636→    for (let col = 0; col < 7; col++) {
   637→      if (!started && col < startDow) { html += '<td></td>'; }
   638→      else if (dayNum > lastDay.getDate()) { html += '<td></td>'; }
   639→      else {
   640→        started = true;
   641→        const d = new Date(year, month, dayNum);
   642→        const dk = dateKey(d);
   643→        const isToday = isSameDay(d, today);
   644→        const isWeekend = d.getDay() === 0 || d.getDay() === 6;
   645→        
   646→        let cls = '';
   647→        if (isToday) cls += ' today';
   648→        if (isWeekend) cls += ' weekend';
   649→        
   650→        html += `<td class="${cls}" onclick="mostraGiorno('${dk}')">`;
   651→        html += `<div class="day-num">${dayNum}</div>`;
   652→        
   653→        if (!isWeekend) {
   654→          const ass = stato.assegnazioni[dk] || {};
   655→          // Mostra ingressi per gruppo
   656→          Object.entries(getGruppi()).forEach(([gid, gruppo]) => {
   657→            if (gid === 'J') return;
   658→            let txt = gid + ':';
   659→            let turni = [];
   660→            gruppo.persone.forEach(p => {
   661→              if (ass[p]) {
   662→                if (ass[p].entrata) turni.push(ass[p].entrata);
   663→                else if (ass[p].stato === 'riposo') turni.push('R');
   664→                else if (ass[p].stato === 'reperibile') turni.push('r');
   665→              }
   666→            });
   667→            txt += turni.join('/');
   668→            const colore = turni[0] && TURNI_COLORS[turni[0]] ? TURNI_COLORS[turni[0]].bg : '#f1f5f9';
   669→            const coloreTx = turni[0] && TURNI_COLORS[turni[0]] ? TURNI_COLORS[turni[0]].tx : '#475569';
   670→            html += `<span class="day-shift" style="background:${colore};color:${coloreTx}">${txt}</span>`;
   671→          });
   672→          if (ass['Jolly']) {
   673→            html += `<span class="day-shift" style="background:${TURNI_COLORS.fisso.bg};color:${TURNI_COLORS.fisso.tx}">J:16:30</span>`;
   674→          }
   675→        }
   676→        
   677→        html += '</td>';
   678→        dayNum++;
   679→      }
   680→    }
   681→    html += '</tr>';
   682→  }
   683→  html += '</tbody></table>';
   684→  document.getElementById('calendarioGrid').innerHTML = html;
   685→}

   879→function renderOre() {
   880→  const year = meseCorrente.getFullYear();
   881→  const month = meseCorrente.getMonth();
   882→  document.getElementById('oreMeseHeader').textContent = MESI_IT[month] + ' ' + year;
   883→  
   884→  const firstDay = new Date(year, month, 1);
   885→  const lastDay = new Date(year, month + 1, 0);
   886→  const gruppi = getGruppi();
   887→  
   888→  let html = '';
   889→  
   890→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   891→    gruppo.persone.forEach(persona => {
   892→      let oreLavorate = 0, giorniLavorati = 0, giorniRiposo = 0;
   893→      
   894→      for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   895→        const dk = dateKey(d);
   896→        const dow = d.getDay();
   897→        if (dow === 0 || dow === 6) continue;
   898→        
   899→        const ass = stato.assegnazioni[dk];
   900→        if (!ass || !ass[persona]) continue;
   901→        
   902→        const info = ass[persona];
   903→        if (info.stato === 'lavoro') {
   904→          oreLavorate += info.ore || stato.oreTurno;
   905→          giorniLavorati++;
   906→        } else if (info.stato === 'riposo' || info.stato === 'reperibile') {
   907→          giorniRiposo++;
   908→        }
   909→      }
   910→      
   911→      if (gid === 'J') {
   912→        html += `<div class="ore-row">`;
   913→        html += `<span class="gruppo-badge gruppo-J" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   914→        html += `<div style="flex:1">`;
   915→        html += `<div style="font-size:12px">Fisso + sostituzioni: <strong>${oreLavorate.toFixed(1)}h</strong> (${giorniLavorati}gg)</div>`;
   916→        html += `</div></div>`;
   917→      } else {
   918→        const pct = Math.min(100, Math.round((oreLavorate / stato.oreTarget) * 100));
   919→        const barClass = pct >= 95 ? 'ok' : pct >= 80 ? 'warn' : 'danger';
   920→        const diff = oreLavorate - stato.oreTarget;
   921→        
   922→        html += `<div class="ore-row">`;
   923→        html += `<span class="gruppo-badge gruppo-${gid}" style="width:28px;height:28px;font-size:11px">${persona}</span>`;
   924→        html += `<div style="flex:1">`;
   925→        html += `<div style="display:flex;justify-content:space-between;font-size:12px;margin-bottom:2px">`;
   926→        html += `<span>${giorniLavorati}gg lav, ${giorniRiposo}gg rip</span>`;
   927→        html += `<span style="font-weight:700;color:${barClass==='ok'?'var(--success)':barClass==='warn'?'var(--accent)':'var(--danger)'}">${oreLavorate.toFixed(1)}h ${diff>=0?'✓':'('+diff.toFixed(1)+'h)'}</span>`;
   928→        html += `</div>`;
   929→        html += `<div class="ore-progress"><div class="ore-bar ${barClass}" style="width:${pct}%"></div></div>`;
   930→        html += `</div></div>`;
   931→      }
   932→    });
   933→  });
   934→  
   935→  document.getElementById('oreList').innerHTML = html;
   936→}

   941→function exportExcel() {
   942→  const year = meseCorrente.getFullYear();
   943→  const month = meseCorrente.getMonth();
   944→  const firstDay = new Date(year, month, 1);
   945→  const lastDay = new Date(year, month + 1, 0);
   946→  const gruppi = getGruppi();
   947→  
   948→  // Intestazione
   949→  let csv = 'Data,Giorno,';
   950→  Object.entries(gruppi).forEach(([gid, gruppo]) => {
   951→    gruppo.persone.forEach(p => {
   952→      csv += p + ' Turno,' + p + ' Orario,' + p + ' Ore,';
   953→    });
   954→  });
   955→  csv += 'Presenti 9:30-15:30,\n';
   956→  
   957→  let totali = {};
   958→  Object.values(gruppi).forEach(g => g.persone.forEach(p => totali[p] = 0));
   959→  
   960→  for (let d = new Date(firstDay); d <= lastDay; d.setDate(d.getDate() + 1)) {
   961→    const dk = dateKey(d);
   962→    const dow = d.getDay();
   963→    if (dow === 0 || dow === 6) continue;
   964→    
   965→    const dayName = GIORNI_FULL[dow === 0 ? 6 : dow - 1];
   966→    const ass = stato.assegnazioni[dk] || {};
   967→    
   968→    csv += `${d.getDate()}/${d.getMonth()+1}/${d.getFullYear()},${dayName},`;
   969→    
   970→    Object.entries(gruppi).forEach(([gid, gruppo]) => {
   971→      gruppo.persone.forEach(persona => {
   972→        const info = ass[persona];
   973→        if (!info) { csv += '—,,0,'; return; }
   974→        
   975→        if (info.stato === 'lavoro') {
   976→          csv += `${info.entrata},${info.entrata}-${info.uscita},${info.ore},`;
   977→          totali[persona] += info.ore || stato.oreTurno;
   978→        } else if (info.stato === 'reperibile') {
   979→          csv += 'Reperibile,,0,';
   980→        } else {
   981→          csv += 'Riposo,,0,';
   982→        }
   983→      });
   984→    });
   985→    
   986→    // Conta presenti alle 9:30
   987→    let presenti930 = 0;
   988→    Object.values(ass).forEach(info => {
   989→      if (info.stato === 'lavoro' && info.entrata && info.uscita) {
   990→        const e = tempoAMinuti(info.entrata);
   991→        const u = tempoAMinuti(info.uscita);
   992→        if (570 >= e && 930 < u) presenti930++;
   993→      }
   994→    });
   995→    csv += presenti930 + ',\n';
   996→  }
   997→  
   998→  // Totali
   999→  csv += '\nTOTALI ORE,,';
  1000→  Object.values(gruppi).forEach(g => {
  1001→    g.persone.forEach(p => {
  1002→      csv += ',,' + totali[p].toFixed(1) + ',';
  1003→    });
  1004→  });
  1005→  csv += '\n';
  1006→  
  1007→  csv += 'TARGET,,';
  1008→  Object.values(gruppi).forEach(g => {
  1009→    g.persone.forEach(p => {
  1010→      csv += ',,' + (g.id === 'J' ? '—' : stato.oreTarget) + ',';
  1011→    });
  1012→  });
  1013→  
  1014→  const blob = new Blob(['\ufeff' + csv], {type: 'text/csv;charset=utf-8;'});
  1015→  const url = URL.createObjectURL(blob);
  1016→  const a = document.createElement('a');
  1017→  a.href = url;
  1018→  a.download = `Turnazioni_${MESI_IT[month]}_${year}.csv`;
  1019→  a.click();
  1020→  URL.revokeObjectURL(url);
  1021→}
